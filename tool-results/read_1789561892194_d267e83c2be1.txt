    63→    63→function fmtFecha(ts: number | null | undefined): string {
    64→    64→  if (!ts) return "—";
    65→    65→  try {
    66→    66→    return new Date(ts).toLocaleString("es-PE", {
    67→    67→      timeZone: "America/Lima",
    68→    68→      day: "2-digit", month: "2-digit", year: "numeric",
    69→    69→      hour: "2-digit", minute: "2-digit",
    70→    70→    });
    71→    71→  } catch { return new Date(ts).toLocaleString("es-PE"); }
    72→    72→}
