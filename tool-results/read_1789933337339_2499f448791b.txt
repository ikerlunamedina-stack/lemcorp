    67→function detectarModelo(serie: string): string | null {
    68→  const s = serie.trim().toUpperCase();
    69→  for (const r of REGLAS_PREFIJO) {
    70→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
    71→  }
    72→  return null;
    73→}
    74→
    75→function validarPrefijo(serie: string, prefijo: string): boolean {
    76→  if (!prefijo.trim()) return true;  // sin prefijo = no validar
    77→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
    78→}

    80→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
    81→function detectarDuplicadosEnSistema(
    82→  pistoleoFilas: { valores: string[] }[],
    83→  equipos: { serie: string }[]
    84→): string[] {
    85→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
    86→  const dups: string[] = [];
    87→  for (const f of pistoleoFilas) {
    88→    const s = (f.valores[0] ?? "").trim();
    89→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
    90→  }
    91→  return dups;
    92→}
    93→
    94→/** Series duplicadas dentro del mismo lote actual */
    95→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
    96→  const seen = new Set<string>();
    97→  const dups: string[] = [];
    98→  for (const f of pistoleoFilas) {
    99→    const s = (f.valores[0] ?? "").trim().toLowerCase();
   100→    if (!s) continue;
   101→    if (seen.has(s)) dups.push(f.valores[0].trim());
   102→    else seen.add(s);
   103→  }
   104→  return dups;
   105→}
