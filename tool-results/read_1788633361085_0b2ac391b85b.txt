   547→function SesionSelector() {
   548→  const miembros = useStore((s) => s.miembros);
   549→  const sesionUsuarioId = useStore((s) => s.sesionUsuarioId);
   550→  const iniciarSesion = useStore((s) => s.iniciarSesion);
   551→  const cerrarSesion = useStore((s) => s.cerrarSesion);
   552→  const tienePermiso = useStore((s) => s.tienePermiso);
   553→
   554→  const miembroActual = sesionUsuarioId
   555→    ? miembros.find((m) => m.id === sesionUsuarioId)
   556→    : null;
   557→
   558→  if (miembros.length === 0) {
   559→    return (
   560→      <p className="text-[12px] text-muted-foreground">
   561→        No hay miembros del equipo registrados. Añádelos desde{" "}
   562→        <Link href="/empresa" className="font-semibold text-primary hover:underline">Empresas</Link>.
   563→      </p>
   564→    );
   565→  }
   566→
   567→  return (
   568→    <div className="flex flex-col gap-3">
   569→      {/* Estado actual */}
   570→      <div className="flex items-center gap-3 rounded-xl border border-border bg-muted/30 p-3">
   571→        <div className={cn(
   572→          "flex h-9 w-9 items-center justify-center rounded-full text-[11px] font-bold",
   573→          !miembroActual ? "bg-primary text-primary-foreground" : "bg-muted text-foreground"
   574→        )}>
   575→          {miembroActual ? miembroActual.nombre.charAt(0).toUpperCase() : "AD"}
   576→        </div>
   577→        <div className="min-w-0 flex-1">
   578→          <p className="text-[12px] font-semibold text-foreground">
   579→            {miembroActual?.nombre || "Admin (dueño)"}
   580→          </p>
   581→          <p className="text-[10px] text-muted-foreground">
   582→            {miembroActual ? ROL_META[miembroActual.rol].label : "Acceso total al sistema"}
   583→          </p>
   584→        </div>
   585→        {miembroActual && (
   586→          <Button variant="outline" size="sm" onClick={cerrarSesion} className="press h-8 rounded-lg">
   587→            <LogOut className="mr-1 h-3.5 w-3.5" /> Cerrar sesión
   588→          </Button>
   589→        )}
   590→      </div>
   591→
   592→      {/* Selector */}
   593→      {!miembroActual && (
   594→        <div>
   595→          <Label className="mb-1.5 block text-[11px] uppercase tracking-wide text-muted-foreground">
   596→            Iniciar sesión como
   597→          </Label>
   598→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   599→            {miembros.map((m) => {
   600→              const esAdmin = m.rol === "administrador";
   601→              return (
   602→                <button
   603→                  key={m.id}
   604→                  onClick={() => iniciarSesion(m.id)}
   605→                  className={cn(
   606→                    "press flex items-center gap-2.5 rounded-xl border border-border bg-card p-2.5 text-left transition-colors hover:bg-accent"
   607→                  )}
   608→                >
   609→                  <div className={cn(
   610→                    "flex h-8 w-8 shrink-0 items-center justify-center rounded-lg text-[11px] font-bold",
   611→                    esAdmin ? "bg-primary text-primary-foreground" : "bg-muted text-muted-foreground"
   612→                  )}>
   613→                    {m.nombre.charAt(0).toUpperCase()}
   614→                  </div>
   615→                  <div className="min-w-0 flex-1">
   616→                    <p className="truncate text-[12px] font-semibold">{m.nombre}</p>
   617→                    <p className="truncate text-[10px] text-muted-foreground">{ROL_META[m.rol].label}</p>

   619→                </button>

   625→          </p>
   626→        </div>
   627→      )}
   628→
   629→      {/* Info de permisos */}
   630→      {miembroActual && (
   631→        <div className="rounded-xl border border-border bg-muted/30 p-3 text-[11px]">
   632→          <p className="mb-1.5 font-semibold text-foreground">Tus permisos efectivos:</p>

   634→            {(Object.keys(PERMISO_META) as Permiso[]).filter((p) => tienePermiso(p)).map((p) => (

   636→                {PERMISO_META[p].label}
