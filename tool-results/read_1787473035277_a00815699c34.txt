   161→function normalizaSerie(s: string): string {
   162→  return s.trim();
   163→}
   164→
   165→function detectarModeloPorPrefijo(serie: string): string | null {
   166→  const s = serie.trim().toUpperCase();
   167→  if (!s) return null;
   168→  for (const regla of REGLAS_PREFIJO) {
   169→    if (s.startsWith(regla.prefijo.toUpperCase())) return regla.modelo;
   170→  }
   171→  return null;
   172→}

   174→export const useStore = create<StoreState>()(
   175→  persist(
   176→    (set, get) => ({
   177→      products: [],
   178→      equipos: [],
   179→      entradas: [],
   180→      despachos: [],
   181→      notas: [],
   182→      recordatorios: [],
   183→      notificaciones: [],
   184→      miembros: [],
   185→      empresa: { ...DEFAULT_EMPRESA },
   186→      settings: { ...DEFAULT_SETTINGS },
   187→
   188→      horario: [],
   189→      memoriaIA: [],
   190→
   191→      activeView: "dashboard",
   192→
   193→      // pistoleo
   194→      pistoleoCampo: "serie",
   195→      pistoleoModelo: "",
   196→      pistoleoEstado: "disponible",
   197→      pistoleoFilas: [],
   198→
   199→      // ─── Navegación ───
   200→      setActiveView: (v) => set({ activeView: v }),
   201→
   202→      // ─── Pistoleo ───
   203→      setPistoleoConfig: (patch) => set({ ...patch }),
   204→      addPistoleoFila: (valores) =>
   205→        set({
   206→          pistoleoFilas: [
   207→            { id: uid(), valores: valores.map((v) => v.trim()), timestamp: Date.now() },
   208→            ...get().pistoleoFilas,
   209→          ],
   210→        }),
   211→      deletePistoleoFila: (id) =>
   212→        set({ pistoleoFilas: get().pistoleoFilas.filter((f) => f.id !== id) }),
   213→      clearPistoleoFilas: () => set({ pistoleoFilas: [] }),
   214→
   215→      confirmarPistoleo: () => {
   216→        const filas = get().pistoleoFilas;
   217→        if (filas.length === 0) return { ok: false, msg: "No hay series para guardar.", count: 0 };
   218→        const { pistoleoModelo, pistoleoEstado } = get();
   219→        let count = 0;
   220→        const nuevos: Equipment[] = [];
   221→        const existentes = new Set(get().equipos.map((e) => e.serie.trim().toLowerCase()));
   222→        const fechasNow = Date.now();
   223→        for (const f of filas) {
   224→          const serie = (f.valores[0] ?? "").trim();
   225→          if (!serie) continue;
   226→          if (existentes.has(serie.toLowerCase())) continue;
   227→          const modelo = pistoleoModelo.trim() || detectarModeloPorPrefijo(serie) || "SIN MODELO";
   228→          nuevos.push({
   229→            id: uid(),
   230→            serie,
   231→            modelo,
   232→            estado: pistoleoEstado,
   233→            ubicacion: "Almacén HUB",
   234→            createdAt: fechasNow,
   235→            updatedAt: fechasNow,
   236→          });
   237→          existentes.add(serie.toLowerCase());
   238→          count++;
   239→        }
   240→        if (count === 0) {
   241→          return { ok: false, msg: "Todas las series ya estaban registradas.", count: 0 };
   242→        }
   243→        set({ equipos: [...nuevos, ...get().equipos], pistoleoFilas: [] });
   244→        return {
   245→          ok: true,
   246→          msg: `${count} equipo(s) guardado(s) correctamente.`,
   247→          count,
   248→        };
   249→      },
   250→
   251→      // ─── Inventario ───
   252→      findProductBySku: (sku) => {
   253→        const norm = sku.trim().toLowerCase();
   254→        return get().products.find((p) => p.sku.trim().toLowerCase() === norm) ?? null;
   255→      },
   256→
   257→      addProduct: (sku, name, quantity, minStock, udm) => {
   258→        const skuTrim = sku.trim();
   259→        if (!skuTrim || !name.trim()) return null;
   260→        if (get().findProductBySku(skuTrim)) return null;
   261→        const p: Product = {
   262→          id: uid(),
   263→          sku: skuTrim,
   264→          name: name.trim(),
   265→          quantity: quantity || 0,
   266→          minStock,
   267→          udm,
   268→          createdAt: Date.now(),
   269→          updatedAt: Date.now(),
   270→        };
   271→        set({ products: [...get().products, p] });
   272→        return p.id;
   273→      },
   274→
   275→      updateProduct: (id, data) =>
   276→        set({
   277→          products: get().products.map((p) =>
   278→            p.id === id ? { ...p, ...data, updatedAt: Date.now() } : p
   279→          ),
   280→        }),
   281→
   282→      deleteProduct: (id) =>
   283→        set({ products: get().products.filter((p) => p.id !== id) }),
   284→
   285→      importProductsBulk: (items) => {
   286→        let ok = 0;
   287→        let dup = 0;
   288→        const nuevos: Product[] = [];
   289→        const existSkus = new Set(get().products.map((p) => p.sku.trim().toLowerCase()));
   290→        for (const it of items) {
   291→          const sku = it.sku.trim();
   292→          if (!sku || !it.name.trim()) continue;
   293→          if (existSkus.has(sku.toLowerCase())) { dup++; continue; }
   294→          existSkus.add(sku.toLowerCase());
   295→          nuevos.push({
   296→            id: uid(),
   297→            sku,
   298→            name: it.name.trim(),
   299→            quantity: it.quantity || 0,
   300→            minStock: it.minStock,
   301→            udm: it.udm,
   302→            createdAt: Date.now(),
   303→            updatedAt: Date.now(),
   304→          });
   305→          ok++;
   306→        }
   307→        if (nuevos.length > 0) set({ products: [...get().products, ...nuevos] });
   308→        return { ok, dup };
   309→      },
   310→
   311→      // ─── Entradas (formato SKU*cantidad por línea) ───
   312→      registrarEntrada: (input) => {
   313→        const lines = input.split("\n").filter((l) => l.trim());
   314→        let count = 0;
   315→        const newEntradas: Entrada[] = [];
   316→        for (const line of lines) {
   317→          const parts = line.trim().split("*");
   318→          if (parts.length < 2) continue;
   319→          const sku = parts[0].trim();
   320→          const cantidad = parseInt(parts[1].trim(), 10);
   321→          if (!sku || isNaN(cantidad) || cantidad <= 0) continue;
   322→          const product = get().findProductBySku(sku);
   323→          if (product) {
   324→            set({
   325→              products: get().products.map((p) =>
   326→                p.id === product.id
   327→                  ? { ...p, quantity: p.quantity + cantidad, updatedAt: Date.now() }
   328→                  : p
   329→              ),
   330→            });
   331→          }
   332→          newEntradas.push({
   333→            id: uid(),
   334→            fecha: Date.now(),
   335→            sku,
   336→            producto: product?.name ?? sku,
   337→            cantidad,
   338→          });
   339→          count++;
   340→        }
   341→        if (count > 0) set({ entradas: [...newEntradas, ...get().entradas] });
   342→        return {
   343→          ok: count > 0,
   344→          msg: count > 0
   345→            ? `${count} entrada(s) registrada(s)`
   346→            : "Formato incorrecto. Usa: SKU*cantidad (ej: 1066990*100)",
   347→          count,
   348→        };
   349→      },
   350→
   351→      deleteEntrada: (id) => {
   352→        const ent = get().entradas.find((e) => e.id === id);
   353→        if (!ent) return;
   354→        const product = get().findProductBySku(ent.sku);
   355→        if (product) {
   356→          set({
   357→            products: get().products.map((p) =>
   358→              p.id === product.id
   359→                ? { ...p, quantity: Math.max(0, p.quantity - ent.cantidad), updatedAt: Date.now() }
   360→                : p
   361→            ),
   362→          });
   363→        }
   364→        set({ entradas: get().entradas.filter((e) => e.id !== id) });
   365→      },
   366→
   367→      // ─── Despachos (valida SKU y stock, descuenta inventario) ───
   368→      registrarDespacho: ({ sku, cantidad, tecnico, destino, observacion, fecha }) => {
   369→        const skuTrim = sku.trim();
   370→        const product = get().findProductBySku(skuTrim);
   371→        if (!product) {
   372→          return { ok: false, msg: `SKU "${skuTrim}" no encontrado en el catálogo.` };
   373→        }
   374→        if (!Number.isFinite(cantidad) || cantidad <= 0) {
   375→          return { ok: false, msg: "La cantidad debe ser mayor a 0." };
   376→        }
   377→        if (product.quantity < cantidad) {
   378→          return {
   379→            ok: false,
   380→            msg: `Stock insuficiente. Disponible: ${product.quantity}, solicitado: ${cantidad}.`,
   381→          };
   382→        }
   383→        const d: Despacho = {
   384→          id: uid(),
   385→          fecha: fecha || Date.now(),
   386→          sku: product.sku,
   387→          producto: product.name,
   388→          cantidad,
   389→          tecnico: tecnico?.trim() || undefined,
   390→          destino: destino?.trim() || undefined,
   391→          observacion: observacion?.trim() || undefined,
   392→        };
   393→        set({
   394→          despachos: [d, ...get().despachos],
   395→          products: get().products.map((p) =>
   396→            p.id === product.id
   397→              ? { ...p, quantity: p.quantity - cantidad, updatedAt: Date.now() }
   398→              : p
   399→          ),
   400→        });
   401→        return {
   402→          ok: true,
   403→          msg: `Despacho registrado: ${cantidad} × ${product.name} → stock: ${product.quantity - cantidad}`,
   404→        };
   405→      },
   406→
   407→      // Registro masivo de despachos (para Excel importado)
   408→      registrarDespachosBulk: (inputDespachos) => {
   409→        let ok = 0;
   410→        let fail = 0;
   411→        let totalUnidades = 0;
   412→        const fails: string[] = [];
   413→        const nuevosDespachos: Despacho[] = [];
   414→        const productosActualizados = new Map<string, number>();
   415→
   416→        for (const item of inputDespachos) {
   417→          const skuTrim = item.sku.trim();
   418→          const product = get().findProductBySku(skuTrim);
   419→          if (!product) {
   420→            fail++;
   421→            fails.push(`SKU "${skuTrim}" no encontrado`);
   422→            continue;
   423→          }
   424→          if (!Number.isFinite(item.cantidad) || item.cantidad <= 0) {
   425→            fail++;
   426→            fails.push(`Cantidad inválida para ${skuTrim}`);
   427→            continue;
   428→          }
   429→          // Calcular stock disponible considerando despachos anteriores del mismo bulk
   430→          const stockActual = productosActualizados.get(product.id) ?? product.quantity;
   431→          if (stockActual < item.cantidad) {
   432→            fail++;
   433→            fails.push(`Stock insuficiente para ${product.name} (disp: ${stockActual}, solicitado: ${item.cantidad})`);
   434→            continue;
   435→          }
   436→          productosActualizados.set(product.id, stockActual - item.cantidad);
   437→          nuevosDespachos.push({
   438→            id: uid(),
   439→            fecha: item.fecha || Date.now(),
   440→            sku: product.sku,
   441→            producto: product.name,
   442→            cantidad: item.cantidad,
   443→            tecnico: item.tecnico?.trim() || undefined,
   444→            destino: item.destino?.trim() || undefined,
   445→            observacion: item.observacion?.trim() || undefined,
   446→          });
   447→          ok++;
   448→          totalUnidades += item.cantidad;
   449→        }
   450→
   451→        if (nuevosDespachos.length > 0) {
   452→          // Ordenar por fecha (más reciente primero)
   453→          nuevosDespachos.sort((a, b) => b.fecha - a.fecha);
   454→          set({
   455→            despachos: [...nuevosDespachos, ...get().despachos],
   456→            products: get().products.map((p) => {
   457→              const newQty = productosActualizados.get(p.id);
   458→              return newQty !== undefined ? { ...p, quantity: newQty, updatedAt: Date.now() } : p;
   459→            }),
   460→          });
   461→        }
   462→        return { ok, fail, fails, totalUnidades };
   463→      },
   464→
   465→      deleteDespacho: (id) => {
   466→        const d = get().despachos.find((x) => x.id === id);
   467→        if (!d) return;
   468→        const product = get().findProductBySku(d.sku);
   469→        if (product) {
   470→          set({
   471→            products: get().products.map((p) =>
   472→              p.id === product.id
   473→                ? { ...p, quantity: p.quantity + d.cantidad, updatedAt: Date.now() }
   474→                : p
   475→            ),
   476→          });
   477→        }
   478→        set({ despachos: get().despachos.filter((x) => x.id !== id) });
   479→      },
   480→
   481→      // ─── Equipos ───
   482→      findEquipmentBySerie: (serie) => {
   483→        const norm = serie.trim().toLowerCase();
   484→        return get().equipos.find((e) => e.serie.trim().toLowerCase() === norm) ?? null;
   485→      },
   486→
   487→      addEquipment: (e) => {
   488→        const serieTrim = e.serie.trim();
   489→        const modeloTrim = e.modelo.trim();
   490→        if (!serieTrim || !modeloTrim) return null;
   491→        if (get().findEquipmentBySerie(serieTrim)) return null;
   492→        const eq: Equipment = {
   493→          id: uid(),
   494→          ...e,
   495→          serie: serieTrim,
   496→          modelo: modeloTrim,
   497→          createdAt: Date.now(),
   498→          updatedAt: Date.now(),
   499→        };
   500→        set({ equipos: [eq, ...get().equipos] });
   501→        return eq.id;
   502→      },
   503→
   504→      addEquipmentBulk: ({ series, modelo, estado, ubicacion, observacion }) => {
   505→        const modeloTrim = modelo.trim();
   506→        let ok = 0;
   507→        let dup = 0;
   508→        const nuevos: Equipment[] = [];
   509→        const existentes = new Set(get().equipos.map((e) => e.serie.trim().toLowerCase()));
   510→        const ts = Date.now();
   511→        for (const raw of series) {
   512→          const s = normalizaSerie(raw);
   513→          if (!s) continue;
   514→          if (existentes.has(s.toLowerCase())) { dup++; continue; }
   515→          existentes.add(s.toLowerCase());
   516→          nuevos.push({
   517→            id: uid(),
   518→            serie: s,
   519→            modelo: modeloTrim,
   520→            estado,
   521→            ubicacion,
   522→            observacion,
   523→            createdAt: ts,
   524→            updatedAt: ts,
   525→          });
   526→          ok++;
   527→        }
   528→        if (nuevos.length > 0) set({ equipos: [...nuevos, ...get().equipos] });
   529→        return { ok, dup };
   530→      },
   531→
   532→      updateEquipment: (id, data) =>
   533→        set({
   534→          equipos: get().equipos.map((e) =>
   535→            e.id === id ? { ...e, ...data, updatedAt: Date.now() } : e
   536→          ),
   537→        }),
   538→
   539→      deleteEquipment: (id) =>
   540→        set({ equipos: get().equipos.filter((e) => e.id !== id) }),
   541→
   542→      deleteEquipmentBulk: (ids) =>
   543→        set({ equipos: get().equipos.filter((e) => !ids.includes(e.id)) }),
   544→
