    46→const STORAGE_KEY = "nuclon-ia-chat-v2";
    47→const CINCO_HORAS = 5 * 60 * 60 * 1000; // 5 horas en ms
    48→
    49→function loadChat(): ChatMsg[] {
    50→  try {
    51→    const raw = localStorage.getItem(STORAGE_KEY);
    52→    if (!raw) return [];
    53→    const parsed = JSON.parse(raw);
    54→    if (!parsed || typeof parsed.timestamp !== "number") return [];
    55→    // Auto-clear después de 5 horas
    56→    if (Date.now() - parsed.timestamp > CINCO_HORAS) {
    57→      localStorage.removeItem(STORAGE_KEY);
    58→      return [];
    59→    }
    60→    if (Array.isArray(parsed.messages)) return parsed.messages;
    61→    return [];
    62→  } catch {
    63→    return [];
    64→  }
    65→}
    66→
    67→function saveChat(msgs: ChatMsg[]) {
    68→  try {
    69→    const toSave = msgs.slice(-60);
    70→    localStorage.setItem(STORAGE_KEY, JSON.stringify({ timestamp: Date.now(), messages: toSave }));
    71→  } catch {
    72→    /* ignore */
    73→  }
    74→}
    75→
    76→function timeAgo(ts: number): string {
    77→  const diff = Date.now() - ts;
    78→  const mins = Math.floor(diff / 60000);
    79→  if (mins < 1) return "ahora";
    80→  if (mins < 60) return `hace ${mins}m`;
    81→  const hrs = Math.floor(mins / 60);
    82→  if (hrs < 24) return `hace ${hrs}h`;
    83→  return new Date(ts).toLocaleDateString("es-PE", { day: "2-digit", month: "2-digit" });
    84→}
