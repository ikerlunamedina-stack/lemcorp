    61→    61→// Sonido de error (beep) para rechazos al pistolear — usa Web Audio API (no MP3)
    62→    62→let audioCtx: AudioContext | null = null;
    63→    63→function playBeepError() {
    64→    64→  try {
    65→    65→    if (!audioCtx) {
    66→    66→      audioCtx = new (window.AudioContext || (window as any).webkitAudioContext)();
    67→    67→    }
    68→    68→    // Si el contexto está suspendido (autoplay policy), reanudar
    69→    69→    if (audioCtx.state === "suspended") {
    70→    70→      audioCtx.resume();
    71→    71→    }
    72→    72→    // Crear dos beeps cortos (estilo error de scanner)
    73→    73→    const now = audioCtx.currentTime;
    74→    74→    // Primer beep (agudo)
    75→    75→    const osc1 = audioCtx.createOscillator();
    76→    76→    const gain1 = audioCtx.createGain();
    77→    77→    osc1.frequency.value = 880; // La5
    78→    78→    osc1.type = "square";
    79→    79→    gain1.gain.setValueAtTime(0.3, now);
    80→    80→    gain1.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
    81→    81→    osc1.connect(gain1);
    82→    82→    gain1.connect(audioCtx.destination);
    83→    83→    osc1.start(now);
    84→    84→    osc1.stop(now + 0.15);
    85→    85→    // Segundo beep (grave) — medio segundo después
    86→    86→    const osc2 = audioCtx.createOscillator();
    87→    87→    const gain2 = audioCtx.createGain();
    88→    88→    osc2.frequency.value = 440; // La4
    89→    89→    osc2.type = "square";
    90→    90→    gain2.gain.setValueAtTime(0.3, now + 0.2);
    91→    91→    gain2.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
    92→    92→    osc2.connect(gain2);
    93→    93→    gain2.connect(audioCtx.destination);
    94→    94→    osc2.start(now + 0.2);
    95→    95→    osc2.stop(now + 0.35);
    96→    96→  } catch {}
    97→    97→}

   107→   107→function detectarModelo(serie: string): string | null {
   108→   108→  const s = serie.trim().toUpperCase();
   109→   109→  for (const r of REGLAS_PREFIJO) {
   110→   110→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
   111→   111→  }
   112→   112→  return null;
   113→   113→}

   115→   115→function validarPrefijo(serie: string, prefijo: string): boolean {
   116→   116→  if (!prefijo.trim()) return true;  // sin prefijo = no validar
   117→   117→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
   118→   118→}

   120→   120→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
   121→   121→function detectarDuplicadosEnSistema(
   122→   122→  pistoleoFilas: { valores: string[] }[],
   123→   123→  equipos: { serie: string }[]
   124→   124→): string[] {
   125→   125→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
   126→   126→  const dups: string[] = [];
   127→   127→  for (const f of pistoleoFilas) {
   128→   128→    const s = (f.valores[0] ?? "").trim();
   129→   129→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
   130→   130→  }
   131→   131→  return dups;
   132→   132→}
   133→   133→
   134→   134→/** Series duplicadas dentro del mismo lote actual */
   135→   135→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
   136→   136→  const seen = new Set<string>();
   137→   137→  const dups: string[] = [];
   138→   138→  for (const f of pistoleoFilas) {
   139→   139→    const s = (f.valores[0] ?? "").trim().toLowerCase();
   140→   140→    if (!s) continue;
   141→   141→    if (seen.has(s)) dups.push(f.valores[0].trim());
   142→   142→    else seen.add(s);
   143→   143→  }
   144→   144→  return dups;
   145→   145→}
