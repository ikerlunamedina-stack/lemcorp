    32→function useCountUp(target: number, duration = 800) {
    33→  const [val, setVal] = useState(0);
    34→  useEffect(() => {
    35→    let start = 0;
    36→    const startTime = Date.now();
    37→    const tick = () => {
    38→      const elapsed = Date.now() - startTime;
    39→      const progress = Math.min(elapsed / duration, 1);
    40→      const eased = 1 - Math.pow(1 - progress, 3);
    41→      setVal(Math.round(start + (target - start) * eased));
    42→      if (progress < 1) requestAnimationFrame(tick);
    43→    };
    44→    requestAnimationFrame(tick);
    45→  }, [target, duration]);
    46→  return val;
    47→}

    49→/** Mini gráfico de línea SVG (legacy, para compatibilidad) */
    50→function MiniChart({ data, color = "var(--primary)", height = 40 }: {
    51→  data: number[]; color?: string; height?: number;
    52→}) {
    53→  const max = Math.max(...data, 1);
    54→  const min = Math.min(...data, 0);
    55→  const range = max - min || 1;
    56→  const points = data.map((v, i) => {
    57→    const x = (i / Math.max(data.length - 1, 1)) * 100;
    58→    const y = height - ((v - min) / range) * (height - 4) - 2;
    59→    return `${x},${y}`;
    60→  }).join(" ");
    61→  const areaPoints = `0,${height} ${points} 100,${height}`;
    62→  return (
    63→    <svg viewBox={`0 0 100 ${height}`} className="w-full" style={{ height }} preserveAspectRatio="none">
    64→      <polygon points={areaPoints} fill={color} opacity={0.1} />
    65→      <polyline points={points} fill="none" stroke={color} strokeWidth={1.5} strokeLinejoin="round" strokeLinecap="round" />
    66→    </svg>
    67→  );
    68→}

   161→/** Donut chart con conic-gradient */
   162→function Donut({ segments, size = 100 }: {
   163→  segments: { label: string; value: number; color: string }[];
   164→  size?: number;
   165→}) {
   166→  const total = segments.reduce((s, x) => s + x.value, 0) || 1;
   167→  const gradParts: string[] = [];
   168→  segments.reduce((acc, s) => {
   169→    const pct = (s.value / total) * 100;
   170→    const start = acc;
   171→    const end = acc + pct;
   172→    gradParts.push(`${s.color} ${start}% ${end}%`);
   173→    return end;
   174→  }, 0);
   175→  const grad = gradParts.join(", ");
   176→  return (
   177→    <div className="relative" style={{ width: size, height: size }}>
   178→      <div className="rounded-full" style={{
   179→        width: size, height: size,
   180→        background: `conic-gradient(${grad})`,
   181→      }} />
   182→      <div className="absolute inset-[20%] rounded-full bg-card flex items-center justify-center">
   183→        <span className="text-[14px] font-bold tabular-nums text-foreground">{total}</span>
   184→      </div>
   185→    </div>
   186→  );
   187→}
