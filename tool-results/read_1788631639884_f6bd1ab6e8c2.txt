    62→    62→function detectarModelo(serie: string, prefijoEnabled: boolean): string | null {
    63→    63→  if (!prefijoEnabled) return null;
    64→    64→  const s = serie.trim().toUpperCase();
    65→    65→  for (const r of REGLAS_PREFIJO) {
    66→    66→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
    67→    67→  }
    68→    68→  return null;
    69→    69→}
    70→    70→
    71→    71→function validarPrefijo(serie: string, prefijo: string, enabled: boolean): boolean {
    72→    72→  if (!enabled) return true;
    73→    73→  if (!prefijo.trim()) return true;
    74→    74→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
    75→    75→}

    77→    77→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
    78→    78→function detectarDuplicadosEnSistema(
    79→    79→  pistoleoFilas: { valores: string[] }[],
    80→    80→  equipos: { serie: string }[]
    81→    81→): string[] {
    82→    82→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
    83→    83→  const dups: string[] = [];
    84→    84→  for (const f of pistoleoFilas) {
    85→    85→    const s = (f.valores[0] ?? "").trim();
    86→    86→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
    87→    87→  }
    88→    88→  return dups;
    89→    89→}
    90→    90→
    91→    91→/** Series duplicadas dentro del mismo lote actual */
    92→    92→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
    93→    93→  const seen = new Set<string>();
    94→    94→  const dups: string[] = [];
    95→    95→  for (const f of pistoleoFilas) {
    96→    96→    const s = (f.valores[0] ?? "").trim().toLowerCase();
    97→    97→    if (!s) continue;
    98→    98→    if (seen.has(s)) dups.push(f.valores[0].trim());
    99→    99→    else seen.add(s);
   100→   100→  }
   101→   101→  return dups;
   102→   102→}
