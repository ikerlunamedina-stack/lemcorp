    61→// Sonido de error (beep) para rechazos al pistolear — usa Web Audio API (no MP3)
    62→let audioCtx: AudioContext | null = null;
    63→function playBeepError() {
    64→  try {
    65→    if (!audioCtx) {
    66→      audioCtx = new (window.AudioContext || (window as any).webkitAudioContext)();
    67→    }
    68→    // Si el contexto está suspendido (autoplay policy), reanudar
    69→    if (audioCtx.state === "suspended") {
    70→      audioCtx.resume();
    71→    }
    72→    // Crear dos beeps cortos (estilo error de scanner)
    73→    const now = audioCtx.currentTime;
    74→    // Primer beep (agudo)
    75→    const osc1 = audioCtx.createOscillator();
    76→    const gain1 = audioCtx.createGain();
    77→    osc1.frequency.value = 880; // La5
    78→    osc1.type = "square";
    79→    gain1.gain.setValueAtTime(0.3, now);
    80→    gain1.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
    81→    osc1.connect(gain1);
    82→    gain1.connect(audioCtx.destination);
    83→    osc1.start(now);
    84→    osc1.stop(now + 0.15);
    85→    // Segundo beep (grave) — medio segundo después
    86→    const osc2 = audioCtx.createOscillator();
    87→    const gain2 = audioCtx.createGain();
    88→    osc2.frequency.value = 440; // La4
    89→    osc2.type = "square";
    90→    gain2.gain.setValueAtTime(0.3, now + 0.2);
    91→    gain2.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
    92→    osc2.connect(gain2);
    93→    gain2.connect(audioCtx.destination);
    94→    osc2.start(now + 0.2);
    95→    osc2.stop(now + 0.35);
    96→  } catch {}
    97→}

   107→function detectarModelo(serie: string): string | null {
   108→  const s = serie.trim().toUpperCase();
   109→  for (const r of REGLAS_PREFIJO) {
   110→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
   111→  }
   112→  return null;
   113→}
   114→
   115→function validarPrefijo(serie: string, prefijo: string): boolean {
   116→  if (!prefijo.trim()) return true;  // sin prefijo = no validar
   117→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
   118→}

   120→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
   121→function detectarDuplicadosEnSistema(
   122→  pistoleoFilas: { valores: string[] }[],
   123→  equipos: { serie: string }[]
   124→): string[] {
   125→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
   126→  const dups: string[] = [];
   127→  for (const f of pistoleoFilas) {
   128→    const s = (f.valores[0] ?? "").trim();
   129→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
   130→  }
   131→  return dups;
   132→}
   133→
   134→/** Series duplicadas dentro del mismo lote actual */
   135→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
   136→  const seen = new Set<string>();
   137→  const dups: string[] = [];
   138→  for (const f of pistoleoFilas) {
   139→    const s = (f.valores[0] ?? "").trim().toLowerCase();
   140→    if (!s) continue;
   141→    if (seen.has(s)) dups.push(f.valores[0].trim());
   142→    else seen.add(s);
   143→  }
   144→  return dups;
   145→}
