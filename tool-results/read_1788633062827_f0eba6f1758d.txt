    47→const STORAGE_KEY = "nuclon-ia-chat-v2";
    48→const CINCO_HORAS = 5 * 60 * 60 * 1000; // 5 horas en ms
    49→
    50→function loadChat(): ChatMsg[] {
    51→  try {
    52→    const raw = localStorage.getItem(STORAGE_KEY);
    53→    if (!raw) return [];
    54→    const parsed = JSON.parse(raw);
    55→    if (!parsed || typeof parsed.timestamp !== "number") return [];
    56→    // Auto-clear después de 5 horas
    57→    if (Date.now() - parsed.timestamp > CINCO_HORAS) {
    58→      localStorage.removeItem(STORAGE_KEY);
    59→      return [];
    60→    }
    61→    if (Array.isArray(parsed.messages)) return parsed.messages;
    62→    return [];
    63→  } catch {
    64→    return [];
    65→  }
    66→}
    67→
    68→function saveChat(msgs: ChatMsg[]) {
    69→  try {
    70→    const toSave = msgs.slice(-60);
    71→    localStorage.setItem(STORAGE_KEY, JSON.stringify({ timestamp: Date.now(), messages: toSave }));
    72→  } catch {
    73→    /* ignore */
    74→  }
    75→}
    76→
    77→function timeAgo(ts: number): string {
    78→  const diff = Date.now() - ts;
    79→  const mins = Math.floor(diff / 60000);
    80→  if (mins < 1) return "ahora";
    81→  if (mins < 60) return `hace ${mins}m`;
    82→  const hrs = Math.floor(mins / 60);
    83→  if (hrs < 24) return `hace ${hrs}h`;
    84→  return new Date(ts).toLocaleDateString("es-PE", { day: "2-digit", month: "2-digit" });
    85→}
