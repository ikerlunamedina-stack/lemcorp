    62→function detectarModelo(serie: string, prefijoEnabled: boolean): string | null {
    63→  if (!prefijoEnabled) return null;
    64→  const s = serie.trim().toUpperCase();
    65→  for (const r of REGLAS_PREFIJO) {
    66→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
    67→  }
    68→  return null;
    69→}
    70→
    71→function validarPrefijo(serie: string, prefijo: string, enabled: boolean): boolean {
    72→  if (!enabled) return true;
    73→  if (!prefijo.trim()) return true;
    74→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
    75→}

    77→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
    78→function detectarDuplicadosEnSistema(
    79→  pistoleoFilas: { valores: string[] }[],
    80→  equipos: { serie: string }[]
    81→): string[] {
    82→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
    83→  const dups: string[] = [];
    84→  for (const f of pistoleoFilas) {
    85→    const s = (f.valores[0] ?? "").trim();
    86→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
    87→  }
    88→  return dups;
    89→}
    90→
    91→/** Series duplicadas dentro del mismo lote actual */
    92→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
    93→  const seen = new Set<string>();
    94→  const dups: string[] = [];
    95→  for (const f of pistoleoFilas) {
    96→    const s = (f.valores[0] ?? "").trim().toLowerCase();
    97→    if (!s) continue;
    98→    if (seen.has(s)) dups.push(f.valores[0].trim());
    99→    else seen.add(s);
   100→  }
   101→  return dups;
   102→}

   977→                        </p>
   978→                      )}
   979→                    </div>

   982→                        {ESTADO_META[eq.estado].short}

   986→                );
   987→              })}
   988→            </ul>
   989→          </div>
   990→          <DialogFooter>
   991→            <DialogClose asChild>
   992→              <Button variant="outline" className="rounded-xl">Entendido</Button>
   993→            </DialogClose>
   994→          </DialogFooter>
   995→        </DialogContent>
   996→      </Dialog>
   997→    </div>
   998→  );
