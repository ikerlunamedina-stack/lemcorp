   184→function normalizaSerie(s: string): string {
   185→  return s.trim();
   186→}
   187→
   188→function detectarModeloPorPrefijo(serie: string): string | null {
   189→  const s = serie.trim().toUpperCase();
   190→  if (!s) return null;
   191→  for (const regla of REGLAS_PREFIJO) {
   192→    if (s.startsWith(regla.prefijo.toUpperCase())) return regla.modelo;
   193→  }
   194→  return null;
   195→}

   197→export const useStore = create<StoreState>()(
   198→  persist(
   199→    (set, get) => ({
   200→      products: [],
   201→      equipos: [],
   202→      entradas: [],
   203→      despachos: [],
   204→      notas: [],
   205→      recordatorios: [],
   206→      notificaciones: [],
   207→      miembros: [],
   208→      empresa: { ...DEFAULT_EMPRESA },
   209→      settings: { ...DEFAULT_SETTINGS },
   210→
   211→      horario: [],
   212→      memoriaIA: [],
   213→
   214→      bajoStockVisto: 0,
   215→
   216→      sesionUsuarioId: null,
   217→
   218→      activeView: "dashboard",
   219→
   220→      // pistoleo
   221→      pistoleoCampo: "serie",
   222→      pistoleoModelo: "",
   223→      pistoleoEstado: "disponible",
   224→      pistoleoFilas: [],
   225→      pistoleoModeloSeleccionado: "",
   226→
   227→      // ─── Navegación ───
   228→      setActiveView: (v) => set({ activeView: v }),
   229→
   230→      // ─── Notificaciones ───
   231→      marcarBajoStockVisto: (count) => set({ bajoStockVisto: count }),
   232→
   233→      // ─── Sesión y permisos ───
   234→      iniciarSesion: (miembroId) => set({ sesionUsuarioId: miembroId }),
   235→      cerrarSesion: () => set({ sesionUsuarioId: null }),
   236→      tienePermiso: (permiso) => {
   237→        const state = get();
   238→        const userId = state.sesionUsuarioId;
   239→        // Si no hay sesión iniciada, modo ADMIN (dueño) — todos los permisos
   240→        if (!userId) return true;
   241→        const miembro = state.miembros.find((m) => m.id === userId);
   242→        if (!miembro) return true; // fallback admin
   243→        // Admin tiene todo
   244→        if (miembro.rol === "administrador") return true;
   245→        // Permisos del rol
   246→        const permisosRol = PERMISOS_POR_ROL[miembro.rol] ?? [];
   247→        const extra = miembro.permisosExtra ?? [];
   248→        const revocados = miembro.permisosRevocados ?? [];
   249→        const efectivos = new Set([...permisosRol, ...extra]);
   250→        for (const r of revocados) efectivos.delete(r);
   251→        return efectivos.has(permiso);
   252→      },
   253→      setPermisosMiembro: (id, permisosExtra, permisosRevocados) =>
   254→        set({
   255→          miembros: get().miembros.map((m) =>
   256→            m.id === id ? { ...m, permisosExtra, permisosRevocados } : m
   257→          ),
   258→        }),
   259→
   260→      // ─── Pistoleo ───
   261→      setPistoleoConfig: (patch) => set({ ...patch }),
   262→      addPistoleoFila: (valores, modeloSeleccionado) =>
   263→        set({
   264→          pistoleoFilas: [
   265→            {
   266→              id: uid(),
   267→              valores: valores.map((v) => v.trim()),
   268→              timestamp: Date.now(),
   269→              modeloSeleccionado,
   270→            },
   271→            ...get().pistoleoFilas,
   272→          ],
   273→        }),
   274→      updatePistoleoFila: (id, valores, modeloSeleccionado) =>
   275→        set({
   276→          pistoleoFilas: get().pistoleoFilas.map((f) =>
   277→            f.id === id
   278→              ? {
   279→                  ...f,
   280→                  valores: valores.map((v) => v.trim()),
   281→                  modeloSeleccionado: modeloSeleccionado ?? f.modeloSeleccionado,
   282→                }
   283→              : f
   284→          ),
   285→        }),
   286→      deletePistoleoFila: (id) =>
   287→        set({ pistoleoFilas: get().pistoleoFilas.filter((f) => f.id !== id) }),
   288→      clearPistoleoFilas: () => set({ pistoleoFilas: [] }),
   289→
   290→      confirmarPistoleo: () => {
   291→        const filas = get().pistoleoFilas;
   292→        if (filas.length === 0) return { ok: false, msg: "No hay series para guardar.", count: 0 };
   293→        const { pistoleoModelo, pistoleoEstado, pistoleoCampo } = get();
   294→        let count = 0;
   295→        const nuevos: Equipment[] = [];
   296→        const existentes = new Set(get().equipos.map((e) => e.serie.trim().toLowerCase()));
   297→        const duplicadosNoGuardados: string[] = [];
   298→        const fechasNow = Date.now();
   299→        for (const f of filas) {
   300→          const serie = (f.valores[0] ?? "").trim();
   301→          if (!serie) continue;
   302→          if (existentes.has(serie.toLowerCase())) {
   303→            duplicadosNoGuardados.push(serie);
   304→            continue;
   305→          }
   306→          // MAC es valores[1] cuando el modo es serie_mac o serie_mac_cm
   307→          // CM MAC es valores[2] cuando el modo es serie_mac_cm
   308→          const mac = pistoleoCampo === "serie_mac" || pistoleoCampo === "serie_mac_cm"
   309→            ? (f.valores[1] ?? "").trim() || undefined
   310→            : undefined;
   311→          const cmMac = pistoleoCampo === "serie_mac_cm"
   312→            ? (f.valores[2] ?? "").trim() || undefined
   313→            : undefined;
   314→          const modelo = f.modeloSeleccionado?.trim()
   315→            || pistoleoModelo.trim()
   316→            || detectarModeloPorPrefijo(serie)
   317→            || "SIN MODELO";
   318→          nuevos.push({
   319→            id: uid(),
   320→            serie,
   321→            modelo,
   322→            estado: pistoleoEstado,
   323→            ubicacion: "Almacén HUB",
   324→            mac,
   325→            cmMac,
   326→            createdAt: fechasNow,
   327→            updatedAt: fechasNow,
   328→          });
   329→          existentes.add(serie.toLowerCase());
   330→          count++;
   331→        }
   332→        if (count === 0) {
   333→          return {
   334→            ok: false,
   335→            msg: `Las ${duplicadosNoGuardados.length} serie(s) ya estaban registradas en el sistema.`,
   336→            count: 0,
   337→            duplicados: duplicadosNoGuardados,
   338→          };
   339→        }
   340→        set({ equipos: [...nuevos, ...get().equipos], pistoleoFilas: [] });
   341→        return {
   342→          ok: true,
   343→          msg: duplicadosNoGuardados.length > 0
   344→            ? `${count} equipo(s) guardado(s). ${duplicadosNoGuardados.length} ya estaban registradas.`
   345→            : `${count} equipo(s) guardado(s) correctamente.`,
   346→          count,
   347→          duplicados: duplicadosNoGuardados.length > 0 ? duplicadosNoGuardados : undefined,
   348→        };
   349→      },
   350→
   351→      // ─── Inventario ───
   352→      findProductBySku: (sku) => {
   353→        const norm = sku.trim().toLowerCase();
   354→        return get().products.find((p) => p.sku.trim().toLowerCase() === norm) ?? null;
   355→      },
   356→
   357→      addProduct: (sku, name, quantity, minStock, udm) => {
   358→        const skuTrim = sku.trim();
   359→        if (!skuTrim || !name.trim()) return null;
   360→        if (get().findProductBySku(skuTrim)) return null;
   361→        const p: Product = {
   362→          id: uid(),
   363→          sku: skuTrim,
   364→          name: name.trim(),
   365→          quantity: quantity || 0,
   366→          minStock,
   367→          udm,
   368→          createdAt: Date.now(),
   369→          updatedAt: Date.now(),
   370→        };
   371→        set({ products: [...get().products, p] });
   372→        return p.id;
   373→      },
   374→
   375→      updateProduct: (id, data) =>
   376→        set({
   377→          products: get().products.map((p) =>
   378→            p.id === id ? { ...p, ...data, updatedAt: Date.now() } : p
   379→          ),
   380→        }),
   381→
   382→      deleteProduct: (id) =>
   383→        set({ products: get().products.filter((p) => p.id !== id) }),
   384→
   385→      importProductsBulk: (items) => {
   386→        let ok = 0;
   387→        let dup = 0;
   388→        const nuevos: Product[] = [];
   389→        const existSkus = new Set(get().products.map((p) => p.sku.trim().toLowerCase()));
   390→        for (const it of items) {
   391→          const sku = it.sku.trim();
   392→          if (!sku || !it.name.trim()) continue;
   393→          if (existSkus.has(sku.toLowerCase())) { dup++; continue; }
   394→          existSkus.add(sku.toLowerCase());
   395→          nuevos.push({
   396→            id: uid(),
   397→            sku,
   398→            name: it.name.trim(),
   399→            quantity: it.quantity || 0,
   400→            minStock: it.minStock,
   401→            udm: it.udm,
   402→            createdAt: Date.now(),
   403→            updatedAt: Date.now(),
   404→          });
   405→          ok++;
   406→        }
   407→        if (nuevos.length > 0) set({ products: [...get().products, ...nuevos] });
   408→        return { ok, dup };
   409→      },
   410→
   411→      // ─── Entradas (formato SKU*cantidad por línea) ───
   412→      registrarEntrada: (input) => {
   413→        const lines = input.split("\n").filter((l) => l.trim());
   414→        let count = 0;
   415→        const newEntradas: Entrada[] = [];
   416→        for (const line of lines) {
   417→          const parts = line.trim().split("*");
   418→          if (parts.length < 2) continue;
   419→          const sku = parts[0].trim();
   420→          const cantidad = parseInt(parts[1].trim(), 10);
   421→          if (!sku || isNaN(cantidad) || cantidad <= 0) continue;
   422→          const product = get().findProductBySku(sku);
   423→          if (product) {
   424→            set({
   425→              products: get().products.map((p) =>
   426→                p.id === product.id
   427→                  ? { ...p, quantity: p.quantity + cantidad, updatedAt: Date.now() }
   428→                  : p
   429→              ),
   430→            });
   431→          }
   432→          newEntradas.push({
   433→            id: uid(),
   434→            fecha: Date.now(),
   435→            sku,
   436→            producto: product?.name ?? sku,
   437→            cantidad,
   438→          });
   439→          count++;
   440→        }
   441→        if (count > 0) set({ entradas: [...newEntradas, ...get().entradas] });
   442→        return {
   443→          ok: count > 0,
   444→          msg: count > 0
   445→            ? `${count} entrada(s) registrada(s)`
   446→            : "Formato incorrecto. Usa: SKU*cantidad (ej: 1066990*100)",
   447→          count,
   448→        };
   449→      },
   450→
   451→      deleteEntrada: (id) => {
   452→        const ent = get().entradas.find((e) => e.id === id);
   453→        if (!ent) return;
   454→        const product = get().findProductBySku(ent.sku);
   455→        if (product) {
   456→          set({
   457→            products: get().products.map((p) =>
   458→              p.id === product.id
   459→                ? { ...p, quantity: Math.max(0, p.quantity - ent.cantidad), updatedAt: Date.now() }
   460→                : p
   461→            ),
   462→          });
   463→        }
   464→        set({ entradas: get().entradas.filter((e) => e.id !== id) });
   465→      },
   466→
   467→      // ─── Despachos (valida SKU y stock, descuenta inventario) ───
   468→      registrarDespacho: ({ sku, cantidad, tecnico, destino, observacion, fecha }) => {
   469→        const skuTrim = sku.trim();
   470→        const product = get().findProductBySku(skuTrim);
   471→        if (!product) {
   472→          return { ok: false, msg: `SKU "${skuTrim}" no encontrado en el catálogo.` };
   473→        }
   474→        if (!Number.isFinite(cantidad) || cantidad <= 0) {
   475→          return { ok: false, msg: "La cantidad debe ser mayor a 0." };
   476→        }
   477→        if (product.quantity < cantidad) {
   478→          return {
   479→            ok: false,
   480→            msg: `Stock insuficiente. Disponible: ${product.quantity}, solicitado: ${cantidad}.`,
   481→          };
   482→        }
   483→        const d: Despacho = {
   484→          id: uid(),
   485→          fecha: fecha || Date.now(),
   486→          sku: product.sku,
   487→          producto: product.name,
   488→          cantidad,
   489→          tecnico: tecnico?.trim() || undefined,
   490→          destino: destino?.trim() || undefined,
   491→          observacion: observacion?.trim() || undefined,
   492→        };
   493→        set({
   494→          despachos: [d, ...get().despachos],
   495→          products: get().products.map((p) =>
   496→            p.id === product.id
   497→              ? { ...p, quantity: p.quantity - cantidad, updatedAt: Date.now() }
   498→              : p
   499→          ),
   500→        });
   501→        return {
   502→          ok: true,
   503→          msg: `Despacho registrado: ${cantidad} × ${product.name} → stock: ${product.quantity - cantidad}`,
   504→        };
   505→      },
   506→
   507→      // Registro masivo de despachos (para Excel importado)
   508→      registrarDespachosBulk: (inputDespachos) => {
   509→        let ok = 0;
   510→        let fail = 0;
   511→        let totalUnidades = 0;
   512→        const fails: string[] = [];
   513→        const nuevosDespachos: Despacho[] = [];
   514→        const productosActualizados = new Map<string, number>();
   515→
   516→        for (const item of inputDespachos) {
   517→          const skuTrim = item.sku.trim();
   518→          const product = get().findProductBySku(skuTrim);
   519→          if (!product) {
   520→            fail++;
   521→            fails.push(`SKU "${skuTrim}" no encontrado`);
   522→            continue;
   523→          }
   524→          if (!Number.isFinite(item.cantidad) || item.cantidad <= 0) {
   525→            fail++;
   526→            fails.push(`Cantidad inválida para ${skuTrim}`);
   527→            continue;
   528→          }
   529→          // Calcular stock disponible considerando despachos anteriores del mismo bulk
   530→          const stockActual = productosActualizados.get(product.id) ?? product.quantity;
