     2→     2→

     4→     4→import { useRouter } from "next/navigation";
     5→     5→import {

     7→     7→  AlertTriangle,
     8→     8→  Settings,
     9→     9→  ChevronLeft,
    10→    10→  ChevronRight,

    12→    12→  TrendingUp,

    16→    16→  Check,
    17→    17→  Undo,
    18→    18→  Filter,
    19→    19→  MapPin,
    20→    20→} from "lucide-react";
    21→    21→import { useStore } from "@/lib/store";
    22→    22→import {
    23→    23→  ESTADO_META,
    24→    24→  ANTIGUEDAD_META,
    25→    25→  calcularAntiguedad,
    26→    26→  type EstadoEquipo,
    27→    27→} from "@/lib/types";
    28→    28→import { fmtNum } from "@/lib/num";
    29→    29→import { cn } from "@/lib/utils";
    30→    30→import { AuroraSearchInput } from "@/components/lem/aurora-search-input";
    31→    31→import { EstadoIcon } from "@/components/lem/estado-icon";
    32→    32→
    33→    33→const ICON_PROPS = { strokeWidth: 1.5 } as const;
    34→    34→const ESTADOS: EstadoEquipo[] = ["disponible", "averiado", "en_retiro"];
    35→    35→
    36→    36→export function EquiposView() {
    37→    37→  const equipos = useStore((s) => s.equipos);
    38→    38→  const despachos = useStore((s) => s.despachos);
    39→    39→  const router = useRouter();
    40→    40→
    41→    41→  const [query, setQuery] = useState("");
    42→    42→  const [estadoFilter, setEstadoFilter] = useState<"todos" | EstadoEquipo>("todos");
    43→    43→  const [tab, setTab] = useState<"equipos" | "recomendaciones" | "movimientos">("equipos");
    44→    44→  const [pagina, setPagina] = useState(1);
    45→    45→  const PRODUCTOS_POR_PAGINA = 10;
    46→    46→
    47→    47→  // Resetear página a 1 cuando cambian query/filtro/equipos
    48→    48→  useEffect(() => {
    49→    49→    setPagina(1);
    50→    50→  }, [query, estadoFilter, equipos]);
    51→    51→
    52→    52→  // Filtrado por estado + query
    53→    53→  const filtered = useMemo(
    54→    54→    () =>
    55→    55→      equipos.filter((e) => {
    56→    56→        if (estadoFilter !== "todos" && e.estado !== estadoFilter) return false;
    57→    57→        const q = query.toLowerCase().trim();
    58→    58→        if (!q) return true;
    59→    59→        return (
    60→    60→          e.serie.toLowerCase().includes(q) ||
    61→    61→          e.modelo.toLowerCase().includes(q) ||
    62→    62→          (e.ubicacion ?? "").toLowerCase().includes(q)
    63→    63→        );
    64→    64→      }),
    65→    65→    [equipos, estadoFilter, query],
    66→    66→  );
    67→    67→
    68→    68→  // Agrupar por modelo (ordenado por cantidad descendente)
    69→    69→  const models = useMemo(() => {
    70→    70→    const byModel: Record<string, typeof equipos> = {};
    71→    71→    for (const e of filtered) {
    72→    72→      const k = e.modelo || "Sin modelo";
    73→    73→      if (!byModel[k]) byModel[k] = [];
    74→    74→      byModel[k].push(e);
    75→    75→    }
    76→    76→    return Object.entries(byModel).sort((a, b) => b[1].length - a[1].length);
    77→    77→  }, [filtered]);
    78→    78→
    79→    79→  // Cálculo de paginación
    80→    80→  const totalPaginas = Math.max(1, Math.ceil(models.length / PRODUCTOS_POR_PAGINA));
    81→    81→  const paginaActual = Math.min(pagina, totalPaginas);
    82→    82→  const indiceInicio = (paginaActual - 1) * PRODUCTOS_POR_PAGINA;
    83→    83→  const paginaModels = models.slice(indiceInicio, indiceInicio + PRODUCTOS_POR_PAGINA);
    84→    84→
    85→    85→  // Cantidad de modelos únicos en el catálogo completo
    86→    86→  const modelosDistintos = useMemo(
    87→    87→    () => new Set(equipos.map((e) => e.modelo || "Sin modelo")).size,
    88→    88→    [equipos],
    89→    89→  );
    90→    90→
    91→    91→  // ─── KPIs superiores ───
    92→    92→  const kpis = useMemo(() => {
    93→    93→    const disponibles = equipos.filter((e) => e.estado === "disponible").length;
    94→    94→    const averiadosList = equipos.filter((e) => e.estado === "averiado");
    95→    95→    const enRetiro = equipos.filter((e) => e.estado === "en_retiro").length;
    96→    96→
    97→    97→    // Sin uso 60+d: equipos disponibles hace 60+ días sin despachar
    98→    98→    const ahora = Date.now();
    99→    99→    const dias60 = 60 * 86400_000;
   100→   100→    const sinUsoList = equipos.filter((e) => {
   101→   101→      if (e.estado !== "disponible") return false;
   102→   102→      const seriesEnDespachos = despachos.filter((d) => d.series?.includes(e.serie));
   103→   103→      const ultimoDespacho =
   104→   104→        seriesEnDespachos.length > 0 ? Math.max(...seriesEnDespachos.map((d) => d.fecha)) : 0;
   105→   105→      const referencia = ultimoDespacho || e.createdAt;
   106→   106→      return ahora - referencia >= dias60;
   107→   107→    });
   108→   108→
   109→   109→    return {
   110→   110→      catalogo: equipos.length,
   111→   111→      disponibles,
   112→   112→      averiados: averiadosList.length,
   113→   113→      enRetiro,
   114→   114→      sinUso: sinUsoList.length,
   115→   115→      modelosDistintos,
   116→   116→      averiadosList,
   117→   117→      sinUsoList,
   118→   118→    };
   119→   119→  }, [equipos, despachos, modelosDistintos]);
   120→   120→
   121→   121→  // ─── Recomendaciones Inteligentes ───
   122→   122→  const recomendaciones = useMemo(() => {
   123→   123→    const ahora = Date.now();
   124→   124→
   125→   125→    // 1. Averiados a reparar (con días en estado averiado, usando updatedAt)
   126→   126→    const averiadosConDias = kpis.averiadosList
   127→   127→      .map((e) => ({
   128→   128→        equipo: e,
   129→   129→        dias: Math.max(0, Math.floor((ahora - (e.updatedAt || e.createdAt)) / 86400_000)),
   130→   130→      }))
   131→   131→      .sort((a, b) => b.dias - a.dias);
   132→   132→
   133→   133→    // 2. Sin uso prolongado: disponibles con 60+ días sin despacho
   134→   134→    const sinUsoConDias = kpis.sinUsoList
   135→   135→      .map((e) => {
   136→   136→        const seriesEnDespachos = despachos.filter((d) => d.series?.includes(e.serie));
   137→   137→        const ultimoDespacho =
   138→   138→          seriesEnDespachos.length > 0 ? Math.max(...seriesEnDespachos.map((d) => d.fecha)) : 0;
   139→   139→        const referencia = ultimoDespacho || e.createdAt;
   140→   140→        return {
   141→   141→          equipo: e,
   142→   142→          ultimoDespacho: referencia,
   143→   143→          dias: Math.max(0, Math.floor((ahora - referencia) / 86400_000)),
   144→   144→        };
   145→   145→      })
   146→   146→      .sort((a, b) => b.dias - a.dias);
   147→   147→
   148→   148→    // 3. Tasa de averías por modelo (solo modelos con al menos 1 averiado)
   149→   149→    const porModelo = new Map<string, { modelo: string; total: number; averiados: number }>();
   150→   150→    for (const e of equipos) {
   151→   151→      const k = e.modelo || "Sin modelo";
   152→   152→      const cur = porModelo.get(k) ?? { modelo: k, total: 0, averiados: 0 };
   153→   153→      cur.total += 1;
   154→   154→      if (e.estado === "averiado") cur.averiados += 1;
   155→   155→      porModelo.set(k, cur);
   156→   156→    }
   157→   157→    const tasaAverias = Array.from(porModelo.values())
   158→   158→      .filter((m) => m.averiados > 0)
   159→   159→      .map((m) => ({ ...m, tasa: m.total > 0 ? (m.averiados / m.total) * 100 : 0 }))
   160→   160→      .sort((a, b) => b.tasa - a.tasa)
   161→   161→      .slice(0, 5);
   162→   162→
   163→   163→    // 4. Top modelos más registrados
   164→   164→    const topModelos = Array.from(porModelo.values())
   165→   165→      .sort((a, b) => b.total - a.total)
   166→   166→      .slice(0, 5);
   167→   167→    const maxTopModelos = topModelos.length > 0 ? topModelos[0].total : 1;
   168→   168→
   169→   169→    return { averiadosConDias, sinUsoConDias, tasaAverias, topModelos, maxTopModelos };
   170→   170→  }, [kpis, despachos, equipos]);
   171→   171→
   172→   172→  // Helper: tiempo relativo "hace X días/meses"
   173→   173→  const fmtRelativo = (ts?: number) => {
   174→   174→    if (!ts) return "—";
   175→   175→    const diff = Date.now() - ts;
   176→   176→    const dias = Math.floor(diff / 86400_000);
   177→   177→    if (dias < 1) return "hoy";
   178→   178→    if (dias === 1) return "ayer";
   179→   179→    if (dias < 7) return `hace ${dias} días`;
   180→   180→    if (dias < 30) return `hace ${Math.floor(dias / 7)} sem`;
   181→   181→    const meses = Math.floor(dias / 30);
   182→   182→    return meses === 1 ? "hace 1 mes" : `hace ${meses} meses`;
   183→   183→  };
   184→   184→
   185→   185→  const goSeries = () => router.push("/series");
   186→   186→
   187→   187→  return (
   188→   188→    <div className="select-text cursor-text px-4 py-8 sm:px-6 lg:px-10 anim-fade-in">
   189→   189→      {/* Header */}
   190→   190→      <header className="anim-slide-up mb-8 flex flex-wrap items-end justify-between gap-4">
   191→   191→        <div>
   192→   192→          <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-muted-foreground">
   193→   193→            {equipos.length} equipos · {models.length} modelos
   194→   194→          </p>
   195→   195→          <h1 className="mt-1 text-[28px] font-semibold tracking-tight text-foreground sm:text-[32px]">Equipos</h1>
   196→   196→        </div>
   197→   197→        <AuroraSearchInput
   198→   198→          value={query}
   199→   199→          onChange={setQuery}
   200→   200→          placeholder="Buscar serie…"
   201→   201→          placeholderFocused="Escribe para filtrar series…"
   202→   202→          className="w-48"
   203→   203→          iconSize="h-3.5 w-3.5"
   204→   204→        />
   205→   205→      </header>
   206→   206→
   207→   207→      {/* Pestañas: Equipos | Recomendaciones | Movimientos */}
   208→   208→      <div className="anim-slide-up mb-6 flex items-center gap-1 border-b border-border">
   209→   209→        {([
   210→   210→          ["equipos", "Equipos", equipos.length],
   211→   211→          ["recomendaciones", "Recomendaciones", kpis.averiados + kpis.sinUso + kpis.modelosDistintos],
   212→   212→          ["movimientos", "Movimientos", despachos.length],
   213→   213→        ] as const).map(([key, label, count]) => (
   214→   214→          <button
   215→   215→            key={key}
   216→   216→            onClick={() => setTab(key)}
   217→   217→            className={cn(
   218→   218→              "relative inline-flex h-10 items-center gap-2 px-4 text-[13px] font-medium transition-colors",
   219→   219→              tab === key ? "text-foreground" : "text-muted-foreground hover:text-foreground",
   220→   220→            )}
   221→   221→          >
   222→   222→            {label}
   223→   223→            <span
   224→   224→              className={cn(
   225→   225→                "rounded-full px-1.5 py-0.5 text-[10px] tabular-nums",
   226→   226→                tab === key ? "bg-foreground text-background" : "bg-muted text-muted-foreground",
   227→   227→              )}
   228→   228→            >
   229→   229→              {count}
   230→   230→            </span>
   231→   231→            {tab === key && <span className="absolute inset-x-0 -bottom-px h-0.5 bg-foreground" />}
   232→   232→          </button>
   233→   233→        ))}
   234→   234→      </div>
   235→   235→
   236→   236→      {/* ─── TAB: EQUIPOS (KPIs + chips + lista con paginación) ─── */}
   237→   237→      {tab === "equipos" && (
   238→   238→        <>
   239→   239→          {/* KPIs superiores — entrada escalonada con stagger delay */}
   240→   240→          <section className="mb-6 grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
   241→   241→            {/* 1. Catálogo */}
   242→   242→            <div
   243→   243→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   244→   244→              style={{ animationDelay: "0ms" }}
   245→   245→            >
   246→   246→              <div className="flex items-center gap-1.5">
   247→   247→                <Package className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   248→   248→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Catálogo</p>
   249→   249→              </div>
   250→   250→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(equipos.length)}</p>
   251→   251→            </div>
   252→   252→            {/* 2. Disponibles */}
   253→   253→            <div
   254→   254→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   255→   255→              style={{ animationDelay: "60ms" }}
   256→   256→            >
   257→   257→              <div className="flex items-center gap-1.5">
   258→   258→                <Check className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   259→   259→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Disponibles</p>
   260→   260→              </div>
   261→   261→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(kpis.disponibles)}</p>
   262→   262→            </div>
   263→   263→            {/* 3. Averiados — pulse dot si hay alerta */}
   264→   264→            <div
   265→   265→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   266→   266→              style={{ animationDelay: "120ms" }}
   267→   267→            >
   268→   268→              <div className="flex items-center gap-1.5">
   269→   269→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Averiados</p>
   270→   270→                {kpis.averiados > 0 && <span className="anim-pulse-dot h-2 w-2 rounded-full bg-destructive" />}
   271→   271→              </div>
   272→   272→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(kpis.averiados)}</p>
   273→   273→            </div>
   274→   274→            {/* 4. En retiro */}
   275→   275→            <div
   276→   276→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   277→   277→              style={{ animationDelay: "180ms" }}
   278→   278→            >
   279→   279→              <div className="flex items-center gap-1.5">
   280→   280→                <Undo className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   281→   281→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">En retiro</p>
   282→   282→              </div>
   283→   283→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(kpis.enRetiro)}</p>
   284→   284→            </div>
   285→   285→            {/* 5. Sin uso 60+d — pulse dot si hay alerta */}
   286→   286→            <div
   287→   287→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   288→   288→              style={{ animationDelay: "240ms" }}
   289→   289→            >
   290→   290→              <div className="flex items-center gap-1.5">
   291→   291→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Sin uso 60+d</p>
   292→   292→                {kpis.sinUso > 0 && <span className="anim-pulse-dot h-2 w-2 rounded-full bg-amber-500" />}
   293→   293→              </div>
   294→   294→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(kpis.sinUso)}</p>
   295→   295→            </div>
   296→   296→            {/* 6. Modelos */}
   297→   297→            <div
   298→   298→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-4 text-card-foreground shadow transition-shadow hover:shadow-md"
   299→   299→              style={{ animationDelay: "300ms" }}
   300→   300→            >
   301→   301→              <div className="flex items-center gap-1.5">
   302→   302→                <Boxes className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   303→   303→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Modelos</p>
   304→   304→              </div>
   305→   305→              <p className="mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(kpis.modelosDistintos)}</p>
   306→   306→            </div>
   307→   307→          </section>
   308→   308→
   309→   309→          {/* Chips de filtro por estado */}
   310→   310→          <div className="anim-slide-up mb-6 flex flex-wrap items-center gap-1">
   311→   311→            <Filter className="mr-1 h-3.5 w-3.5 text-muted-foreground" {...ICON_PROPS} />
   312→   312→            <FilterChip active={estadoFilter === "todos"} onClick={() => setEstadoFilter("todos")} label="Todos" count={equipos.length} />
   313→   313→            {ESTADOS.map((est) => {
   314→   314→              const n = equipos.filter((e) => e.estado === est).length;
   315→   315→              return <FilterChip key={est} active={estadoFilter === est} onClick={() => setEstadoFilter(est)} label={ESTADO_META[est].short} count={n} />;
   316→   316→            })}
   317→   317→          </div>
   318→   318→
   319→   319→          {/* Lista de modelos con paginación */}
   320→   320→          {models.length === 0 ? (
   321→   321→            <p className="py-12 text-center text-[13px] text-muted-foreground">
   322→   322→              {equipos.length === 0 ? "No hay equipos registrados. Los equipos se añaden automáticamente al recibir una guía de remisión SUNAT en la página de Recepciones." : "Sin coincidencias."}
   323→   323→            </p>
   324→   324→          ) : (
   325→   325→            <>
   326→   326→              <div className="divide-y divide-border border-y border-border">
   327→   327→                {paginaModels.map(([modelo, items], idx) => {
   328→   328→                  // ─── Antigüedad: contar equipos disponibles por nivel ───
   329→   329→                  const disp = items.filter((e) => e.estado === "disponible");
   330→   330→                  const ant = disp.map((e) => calcularAntiguedad(e.createdAt));
   331→   331→                  const antAdv = ant.filter((a) => a.nivel === "advertencia").length;
   332→   332→                  const antCrit = ant.filter((a) => a.nivel === "critico").length;
   333→   333→                  const antOld = ant.filter((a) => a.nivel === "antiguo").length;
   334→   334→                  const antTotal = antAdv + antCrit + antOld;
   335→   335→                  return (
   336→   336→                    <div
   337→   337→                      key={modelo}
   338→   338→                      className="anim-fade-in flex items-center gap-4 py-4"
   339→   339→                      style={{ animationDelay: `${idx * 25}ms` }}
   340→   340→                    >
   341→   341→                      <div className="min-w-0 flex-1">
   342→   342→                        <p className="text-[15px] font-medium text-foreground">{modelo}</p>
   343→   343→                        <p className="mt-0.5 text-[12px] text-muted-foreground">
   344→   344→                          {items.length} equipo(s) · {disp.length} disponibles
   345→   345→                          {items.some((e) => e.ubicacion) && (
   346→   346→                            <span className="ml-1.5 inline-flex items-center gap-1 text-[11px]">
   347→   347→                              <MapPin className="h-3 w-3" {...ICON_PROPS} />
   348→   348→                              {[...new Set(items.map((e) => e.ubicacion).filter(Boolean))].slice(0, 3).join(", ")}
   349→   349→                              {[...new Set(items.map((e) => e.ubicacion).filter(Boolean))].length > 3 && "…"}
   350→   350→                            </span>
   351→   351→                          )}
   352→   352→                          {antTotal > 0 && (
   353→   353→                            <span className="ml-1.5 inline-flex items-center gap-1 text-[11px]">
   354→   354→                              <Clock className="h-3 w-3" {...ICON_PROPS} />
   355→   355→                              {antTotal} con antigüedad
   356→   356→                            </span>
   357→   357→                          )}
   358→   358→                        </p>
   359→   359→                      </div>
   360→   360→                      <div className="hidden items-center gap-3 sm:flex">
   361→   361→                        {ESTADOS.map((est) => {
   362→   362→                          const n = items.filter((e) => e.estado === est).length;
   363→   363→                          if (n === 0) return null;
   364→   364→                          return (
   365→   365→                            <span key={est} className="flex items-center gap-1.5 text-[12px] text-muted-foreground">
   366→   366→                              <EstadoIcon name={ESTADO_META[est].icon} className="h-3 w-3" />
   367→   367→                              {n}
   368→   368→                            </span>
   369→   369→                          );
   370→   370→                        })}
   371→   371→                        {/* Indicadores de antigüedad (solo si hay equipos disponibles) */}
