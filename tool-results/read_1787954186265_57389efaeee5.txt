   185→   185→  return s.trim();
   186→   186→}

   188→   188→function detectarModeloPorPrefijo(serie: string): string | null {
   189→   189→  const s = serie.trim().toUpperCase();
   190→   190→  if (!s) return null;
   191→   191→  for (const regla of REGLAS_PREFIJO) {
   192→   192→    if (s.startsWith(regla.prefijo.toUpperCase())) return regla.modelo;
   193→   193→  }
   194→   194→  return null;
   195→   195→}

   237→   237→        const state = get();

   239→   239→        // Si no hay sesión iniciada, modo ADMIN (dueño) — todos los permisos
   240→   240→        if (!userId) return true;
   241→   241→        const miembro = state.miembros.find((m) => m.id === userId);
   242→   242→        if (!miembro) return true; // fallback admin
   243→   243→        // Admin tiene todo

   245→   245→        // Permisos del rol

   249→   249→        const efectivos = new Set([...permisosRol, ...extra]);

   251→   251→        return efectivos.has(permiso);

   271→   271→            ...get().pistoleoFilas,
   272→   272→          ],
   273→   273→        }),
   274→   274→      updatePistoleoFila: (id, valores, modeloSeleccionado) =>
   275→   275→        set({
   276→   276→          pistoleoFilas: get().pistoleoFilas.map((f) =>
   277→   277→            f.id === id
   278→   278→              ? {
   279→   279→                  ...f,
   280→   280→                  valores: valores.map((v) => v.trim()),
   281→   281→                  modeloSeleccionado: modeloSeleccionado ?? f.modeloSeleccionado,
   282→   282→                }
   283→   283→              : f
   284→   284→          ),
   285→   285→        }),
   286→   286→      deletePistoleoFila: (id) =>
   287→   287→        set({ pistoleoFilas: get().pistoleoFilas.filter((f) => f.id !== id) }),
   288→   288→      clearPistoleoFilas: () => set({ pistoleoFilas: [] }),
   289→   289→
   290→   290→      confirmarPistoleo: () => {
   291→   291→        const filas = get().pistoleoFilas;
   292→   292→        if (filas.length === 0) return { ok: false, msg: "No hay series para guardar.", count: 0 };
   293→   293→        const { pistoleoModelo, pistoleoEstado, pistoleoCampo } = get();
   294→   294→        let count = 0;
   295→   295→        const nuevos: Equipment[] = [];
   296→   296→        const existentes = new Set(get().equipos.map((e) => e.serie.trim().toLowerCase()));
   297→   297→        const duplicadosNoGuardados: string[] = [];
   298→   298→        const fechasNow = Date.now();
   299→   299→        for (const f of filas) {
   300→   300→          const serie = (f.valores[0] ?? "").trim();
   301→   301→          if (!serie) continue;
   302→   302→          if (existentes.has(serie.toLowerCase())) {
   303→   303→            duplicadosNoGuardados.push(serie);
   304→   304→            continue;
   305→   305→          }
   306→   306→          // MAC es valores[1] cuando el modo es serie_mac o serie_mac_cm
   307→   307→          // CM MAC es valores[2] cuando el modo es serie_mac_cm
   308→   308→          const mac = pistoleoCampo === "serie_mac" || pistoleoCampo === "serie_mac_cm"
   309→   309→            ? (f.valores[1] ?? "").trim() || undefined
   310→   310→            : undefined;
   311→   311→          const cmMac = pistoleoCampo === "serie_mac_cm"
   312→   312→            ? (f.valores[2] ?? "").trim() || undefined
   313→   313→            : undefined;
   314→   314→          const modelo = f.modeloSeleccionado?.trim()
   315→   315→            || pistoleoModelo.trim()
   316→   316→            || detectarModeloPorPrefijo(serie)
   317→   317→            || "SIN MODELO";
   318→   318→          nuevos.push({
   319→   319→            id: uid(),
   320→   320→            serie,
   321→   321→            modelo,
   322→   322→            estado: pistoleoEstado,
   323→   323→            ubicacion: "Almacén HUB",
   324→   324→            mac,
   325→   325→            cmMac,
   326→   326→            createdAt: fechasNow,
   327→   327→            updatedAt: fechasNow,
   328→   328→          });
   329→   329→          existentes.add(serie.toLowerCase());
   330→   330→          count++;
   331→   331→        }
   332→   332→        if (count === 0) {
   333→   333→          return {
   334→   334→            ok: false,
   335→   335→            msg: `Las ${duplicadosNoGuardados.length} serie(s) ya estaban registradas en el sistema.`,
   336→   336→            count: 0,
   337→   337→            duplicados: duplicadosNoGuardados,
   338→   338→          };
   339→   339→        }
   340→   340→        set({ equipos: [...nuevos, ...get().equipos], pistoleoFilas: [] });
   341→   341→        return {
   342→   342→          ok: true,
   343→   343→          msg: duplicadosNoGuardados.length > 0
   344→   344→            ? `${count} equipo(s) guardado(s). ${duplicadosNoGuardados.length} ya estaban registradas.`
   345→   345→            : `${count} equipo(s) guardado(s) correctamente.`,
   346→   346→          count,
   347→   347→          duplicados: duplicadosNoGuardados.length > 0 ? duplicadosNoGuardados : undefined,
   348→   348→        };
   349→   349→      },
   350→   350→
   351→   351→      // ─── Inventario ───
   352→   352→      findProductBySku: (sku) => {
   353→   353→        const norm = sku.trim().toLowerCase();
   354→   354→        return get().products.find((p) => p.sku.trim().toLowerCase() === norm) ?? null;
   355→   355→      },
   356→   356→
   357→   357→      addProduct: (sku, name, quantity, minStock, udm) => {
   358→   358→        const skuTrim = sku.trim();
   359→   359→        if (!skuTrim || !name.trim()) return null;
   360→   360→        if (get().findProductBySku(skuTrim)) return null;
   361→   361→        const p: Product = {
   362→   362→          id: uid(),
   363→   363→          sku: skuTrim,
   364→   364→          name: name.trim(),
   365→   365→          quantity: quantity || 0,
   366→   366→          minStock,
   367→   367→          udm,
   368→   368→          createdAt: Date.now(),
   369→   369→          updatedAt: Date.now(),
   370→   370→        };
   371→   371→        set({ products: [...get().products, p] });
   372→   372→        return p.id;
   373→   373→      },
   374→   374→
   375→   375→      updateProduct: (id, data) =>
   376→   376→        set({
   377→   377→          products: get().products.map((p) =>
   378→   378→            p.id === id ? { ...p, ...data, updatedAt: Date.now() } : p
   379→   379→          ),
   380→   380→        }),
   381→   381→
   382→   382→      deleteProduct: (id) =>
   383→   383→        set({ products: get().products.filter((p) => p.id !== id) }),
   384→   384→
   385→   385→      importProductsBulk: (items) => {
   386→   386→        let ok = 0;
   387→   387→        let dup = 0;
   388→   388→        const nuevos: Product[] = [];
   389→   389→        const existSkus = new Set(get().products.map((p) => p.sku.trim().toLowerCase()));
   390→   390→        for (const it of items) {
   391→   391→          const sku = it.sku.trim();
   392→   392→          if (!sku || !it.name.trim()) continue;
   393→   393→          if (existSkus.has(sku.toLowerCase())) { dup++; continue; }
   394→   394→          existSkus.add(sku.toLowerCase());
   395→   395→          nuevos.push({
   396→   396→            id: uid(),
   397→   397→            sku,
   398→   398→            name: it.name.trim(),
   399→   399→            quantity: it.quantity || 0,
   400→   400→            minStock: it.minStock,
   401→   401→            udm: it.udm,
   402→   402→            createdAt: Date.now(),
   403→   403→            updatedAt: Date.now(),
   404→   404→          });
   405→   405→          ok++;
   406→   406→        }
   407→   407→        if (nuevos.length > 0) set({ products: [...get().products, ...nuevos] });
   408→   408→        return { ok, dup };
   409→   409→      },
   410→   410→
   411→   411→      // ─── Entradas (formato SKU*cantidad por línea) ───
   412→   412→      registrarEntrada: (input) => {
   413→   413→        const lines = input.split("\n").filter((l) => l.trim());
   414→   414→        let count = 0;
   415→   415→        const newEntradas: Entrada[] = [];
   416→   416→        for (const line of lines) {
   417→   417→          const parts = line.trim().split("*");
   418→   418→          if (parts.length < 2) continue;
   419→   419→          const sku = parts[0].trim();
   420→   420→          const cantidad = parseInt(parts[1].trim(), 10);
   421→   421→          if (!sku || isNaN(cantidad) || cantidad <= 0) continue;
   422→   422→          const product = get().findProductBySku(sku);
   423→   423→          if (product) {
   424→   424→            set({
   425→   425→              products: get().products.map((p) =>
   426→   426→                p.id === product.id
   427→   427→                  ? { ...p, quantity: p.quantity + cantidad, updatedAt: Date.now() }
   428→   428→                  : p
   429→   429→              ),
   430→   430→            });
   431→   431→          }
   432→   432→          newEntradas.push({
   433→   433→            id: uid(),
   434→   434→            fecha: Date.now(),
   435→   435→            sku,
   436→   436→            producto: product?.name ?? sku,
   437→   437→            cantidad,
   438→   438→          });
   439→   439→          count++;
   440→   440→        }
   441→   441→        if (count > 0) set({ entradas: [...newEntradas, ...get().entradas] });
