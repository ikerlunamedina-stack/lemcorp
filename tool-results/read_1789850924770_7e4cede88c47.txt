    65→function detectarModelo(serie: string): string | null {
    66→  const s = serie.trim().toUpperCase();
    67→  for (const r of REGLAS_PREFIJO) {
    68→    if (s.startsWith(r.prefijo.toUpperCase())) return r.modelo;
    69→  }
    70→  return null;
    71→}
    72→
    73→function validarPrefijo(serie: string, prefijo: string): boolean {
    74→  if (!prefijo.trim()) return true;  // sin prefijo = no validar
    75→  return serie.trim().toUpperCase().startsWith(prefijo.trim().toUpperCase());
    76→}

    78→/** Series ya registradas en el sistema (en equipos) que coinciden con las del lote actual */
    79→function detectarDuplicadosEnSistema(
    80→  pistoleoFilas: { valores: string[] }[],
    81→  equipos: { serie: string }[]
    82→): string[] {
    83→  const existentes = new Set(equipos.map((e) => e.serie.trim().toLowerCase()));
    84→  const dups: string[] = [];
    85→  for (const f of pistoleoFilas) {
    86→    const s = (f.valores[0] ?? "").trim();
    87→    if (s && existentes.has(s.toLowerCase())) dups.push(s);
    88→  }
    89→  return dups;
    90→}
    91→
    92→/** Series duplicadas dentro del mismo lote actual */
    93→function detectarDuplicadosEnLote(pistoleoFilas: { valores: string[] }[]): string[] {
    94→  const seen = new Set<string>();
    95→  const dups: string[] = [];
    96→  for (const f of pistoleoFilas) {
    97→    const s = (f.valores[0] ?? "").trim().toLowerCase();
    98→    if (!s) continue;
    99→    if (seen.has(s)) dups.push(f.valores[0].trim());
   100→    else seen.add(s);
   101→  }
   102→  return dups;
   103→}

  1037→          <DialogFooter className="border-t border-border px-5 py-4 sm:justify-end">
