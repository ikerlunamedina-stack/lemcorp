    45→function startOfDay(d: Date): Date {
    46→  const x = new Date(d);
    47→  x.setHours(0, 0, 0, 0);
    48→  return x;
    49→}

    51→function daysAgo(n: number): number {
    52→  return startOfDay(new Date(Date.now() - n * 86400_000)).getTime();
    53→}

    55→/** Computa estadísticas de consumo (despachos) por SKU en los últimos N días. */
    56→function computeConsumo(desps: DespachoDTO[], dias: number) {
    57→  const desde = daysAgo(dias);
    58→  const porSku = new Map<string, { sku: string; nombre: string; unidades: number; eventos: number; udm?: string }>();
    59→  for (const d of desps) {
    60→    if (d.fecha < desde) continue;
    61→    const cur = porSku.get(d.sku) ?? { sku: d.sku, nombre: d.producto || d.sku, unidades: 0, eventos: 0, udm: undefined };
    62→    cur.unidades += Number(d.cantidad) || 0;
    63→    cur.eventos += 1;
    64→    porSku.set(d.sku, cur);
    65→  }
    66→  // Top consumidores
    67→  const top = [...porSku.values()].sort((a, b) => b.unidades - a.unidades);
    68→  return {
    69→    desde,
    70→    totalEventos: top.reduce((s, x) => s + x.eventos, 0),
    71→    totalUnidades: top.reduce((s, x) => s + x.unidades, 0),
    72→    top,
    73→  };
    74→}
    75→
    76→/** Proyecta cuántas unidades se necesitan para los próximos `diasFuturos` días, basado en consumo histórico. */
    77→function proyectarNecesidades(
    78→  products: ProductDTO[],
    79→  consumoTop: { sku: string; nombre: string; unidades: number; eventos: number }[],
    80→  diasHistoricos: number,
    81→  diasFuturos: number
    82→) {
    83→  const consumoPorSku = new Map(consumoTop.map((c) => [c.sku, c]));
    84→  return products
    85→    .map((p) => {
    86→      const c = consumoPorSku.get(p.sku);
    87→      const consumoDiario = c ? c.unidades / diasHistoricos : 0;
    88→      const proyectado = Math.ceil(consumoDiario * diasFuturos);
    89→      const deficit = Math.max(0, proyectado - p.quantity);
    90→      return {
    91→        sku: p.sku,
    92→        nombre: p.name,
    93→        stock: p.quantity,
    94→        minimo: p.minStock ?? 0,
    95→        consumoDiario: Number(consumoDiario.toFixed(2)),
    96→        proyectado,
    97→        deficit,
    98→        udm: p.udm ?? "",
    99→      };
   100→    })
   101→    .filter((x) => x.deficit > 0 || x.consumoDiario > 0)
   102→    .sort((a, b) => b.deficit - a.deficit);
   103→}

   755→    .replace(/^(qu[eé] es|qu[eé] son|qu[eé] significa|qu[ié]n es|qu[ié]n fue|quien es|quien fue|qu[ié]nes son|d[ií]me|dime qu[eé] es|dime sobre|h[aá]blame de|cu[eé]ntame de|cu[eé]ntame sobre|informaci[oó]n sobre|s[aá]bes de|s[aá]bes sobre|qu[eé] sabes de|qu[eé] sabes sobre|cu[aá]l es la|cu[aá]l es el|d[oó]nde est[aá]|c[oó]mo se|por qu[eé]|cu[aá]ndo|hay)\s*/i, "")
   756→    .replace(/\?/g, "")
   757→    .replace(/^(la|el|los|las|un|una|de|del|al|es|en)\s+/i, "")
   758→    .trim();
   759→
   760→  if (temaBusqueda.length > 2) {
   761→    try {
   762→      const wikiUrl = `https://es.wikipedia.org/w/api.php?action=query&list=search&srsearch=${encodeURIComponent(temaBusqueda)}&format=json&utf8=1&srlimit=1`;
   763→      const wikiRes = await fetch(wikiUrl, { headers: { "User-Agent": "Alana-LEMCORP/1.0" } });
   764→      if (wikiRes.ok) {
   765→        const wikiData = await wikiRes.json();
   766→        const searchResults = wikiData?.query?.search;
   767→        if (searchResults && searchResults.length > 0) {
   768→          const pageTitle = searchResults[0].title;
   769→          const summaryUrl = `https://es.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(pageTitle)}`;
   770→          const summaryRes = await fetch(summaryUrl, { headers: { "User-Agent": "Alana-LEMCORP/1.0" } });
   771→          if (summaryRes.ok) {
   772→            const summaryData = await summaryRes.json();
   773→            const extract = summaryData?.extract;
   774→            if (extract && extract.length > 20) {
   775→              let respuesta = extract;
   776→              if (respuesta.length > 400) respuesta = respuesta.slice(0, 400).trim() + "...";
   777→              return `${respuesta}\n\n(Fuente: Wikipedia)`;
   778→            }
   779→          }
   780→        }
   781→      }
   782→    } catch {}
   783→  }
   784→
   785→  // === 7. RESPUESTA FINAL ===
   786→  return `No encontré información exacta sobre eso, ${nombre}. ¿Puedes reformular la pregunta? También puedo ayudarte con matemáticas ("cuánto es 5 por 3"), el almacén ("qué productos tengo"), o dime "ayuda".`;
   787→}
