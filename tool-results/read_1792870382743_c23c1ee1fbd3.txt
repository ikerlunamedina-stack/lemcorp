     1→"use client";
     2→
     3→import { useMemo, useState, useEffect } from "react";
     4→import { useRouter } from "next/navigation";
     5→import {
     6→  ArrowRight,
     7→  AlertTriangle,
     8→  Settings,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  Clock,
    12→  TrendingUp,
    13→  TrendingDown,
    14→  Package,
    15→  Boxes,
    16→  Check,
    17→  Undo,
    18→  Filter,
    19→  MapPin,
    20→} from "lucide-react";
    21→import { useStore } from "@/lib/store";
    22→import {
    23→  ESTADO_META,
    24→  ANTIGUEDAD_META,
    25→  calcularAntiguedad,
    26→  type EstadoEquipo,
    27→} from "@/lib/types";
    28→import { fmtNum } from "@/lib/num";
    29→import { cn } from "@/lib/utils";
    30→import { AuroraSearchInput } from "@/components/lem/aurora-search-input";
    31→import { EstadoIcon } from "@/components/lem/estado-icon";
    32→import { motion, AnimatePresence } from "framer-motion";
    33→import { useCountUp } from "@/lib/hooks/use-count-up";
    34→
    35→const ICON_PROPS = { strokeWidth: 1.5 } as const;
    36→const ESTADOS: EstadoEquipo[] = ["disponible", "averiado", "en_retiro"];
    37→
    38→export function EquiposView() {
    39→  const equipos = useStore((s) => s.equipos);
    40→  const despachos = useStore((s) => s.despachos);
    41→  const router = useRouter();
    42→
    43→  const [query, setQuery] = useState("");
    44→  const [estadoFilter, setEstadoFilter] = useState<"todos" | EstadoEquipo>("todos");
    45→  const [tab, setTab] = useState<"equipos" | "recomendaciones" | "movimientos">("equipos");
    46→  const [pagina, setPagina] = useState(1);
    47→  const PRODUCTOS_POR_PAGINA = 10;
    48→
    49→  // Resetear página a 1 cuando cambian query/filtro/equipos
    50→  useEffect(() => {
    51→    setPagina(1);
    52→  }, [query, estadoFilter, equipos]);
    53→
    54→  // Filtrado por estado + query
    55→  const filtered = useMemo(
    56→    () =>
    57→      equipos.filter((e) => {
    58→        if (estadoFilter !== "todos" && e.estado !== estadoFilter) return false;
    59→        const q = query.toLowerCase().trim();
    60→        if (!q) return true;
    61→        return (
    62→          e.serie.toLowerCase().includes(q) ||
    63→          e.modelo.toLowerCase().includes(q) ||
    64→          (e.ubicacion ?? "").toLowerCase().includes(q)
    65→        );
    66→      }),
    67→    [equipos, estadoFilter, query],
    68→  );
    69→
    70→  // Agrupar por modelo (ordenado por cantidad descendente)
    71→  const models = useMemo(() => {
    72→    const byModel: Record<string, typeof equipos> = {};
    73→    for (const e of filtered) {
    74→      const k = e.modelo || "Sin modelo";
    75→      if (!byModel[k]) byModel[k] = [];
    76→      byModel[k].push(e);
    77→    }
    78→    return Object.entries(byModel).sort((a, b) => b[1].length - a[1].length);
    79→  }, [filtered]);
    80→
    81→  // Cálculo de paginación
    82→  const totalPaginas = Math.max(1, Math.ceil(models.length / PRODUCTOS_POR_PAGINA));
    83→  const paginaActual = Math.min(pagina, totalPaginas);
    84→  const indiceInicio = (paginaActual - 1) * PRODUCTOS_POR_PAGINA;
    85→  const paginaModels = models.slice(indiceInicio, indiceInicio + PRODUCTOS_POR_PAGINA);
    86→
    87→  // Cantidad de modelos únicos en el catálogo completo
    88→  const modelosDistintos = useMemo(
    89→    () => new Set(equipos.map((e) => e.modelo || "Sin modelo")).size,
    90→    [equipos],
    91→  );
    92→
    93→  // ─── KPIs superiores ───
    94→  const kpis = useMemo(() => {
    95→    const disponibles = equipos.filter((e) => e.estado === "disponible").length;
    96→    const averiadosList = equipos.filter((e) => e.estado === "averiado");
    97→    const enRetiro = equipos.filter((e) => e.estado === "en_retiro").length;
    98→
    99→    // Sin uso 60+d: equipos disponibles hace 60+ días sin despachar
   100→    const ahora = Date.now();
   101→    const dias60 = 60 * 86400_000;
   102→    const sinUsoList = equipos.filter((e) => {
   103→      if (e.estado !== "disponible") return false;
   104→      const seriesEnDespachos = despachos.filter((d) => d.series?.includes(e.serie));
   105→      const ultimoDespacho =
   106→        seriesEnDespachos.length > 0 ? Math.max(...seriesEnDespachos.map((d) => d.fecha)) : 0;
   107→      const referencia = ultimoDespacho || e.createdAt;
   108→      return ahora - referencia >= dias60;
   109→    });
   110→
   111→    return {
   112→      catalogo: equipos.length,
   113→      disponibles,
   114→      averiados: averiadosList.length,
   115→      enRetiro,
   116→      sinUso: sinUsoList.length,
   117→      modelosDistintos,
   118→      averiadosList,
   119→      sinUsoList,
   120→    };
   121→  }, [equipos, despachos, modelosDistintos]);
   122→
   123→  // ─── Recomendaciones Inteligentes ───
   124→  const recomendaciones = useMemo(() => {
   125→    const ahora = Date.now();
   126→
   127→    // 1. Averiados a reparar (con días en estado averiado, usando updatedAt)
   128→    const averiadosConDias = kpis.averiadosList
   129→      .map((e) => ({
   130→        equipo: e,
   131→        dias: Math.max(0, Math.floor((ahora - (e.updatedAt || e.createdAt)) / 86400_000)),
   132→      }))
   133→      .sort((a, b) => b.dias - a.dias);
   134→
   135→    // 2. Sin uso prolongado: disponibles con 60+ días sin despacho
   136→    const sinUsoConDias = kpis.sinUsoList
   137→      .map((e) => {
   138→        const seriesEnDespachos = despachos.filter((d) => d.series?.includes(e.serie));
   139→        const ultimoDespacho =
   140→          seriesEnDespachos.length > 0 ? Math.max(...seriesEnDespachos.map((d) => d.fecha)) : 0;
   141→        const referencia = ultimoDespacho || e.createdAt;
   142→        return {
   143→          equipo: e,
   144→          ultimoDespacho: referencia,
   145→          dias: Math.max(0, Math.floor((ahora - referencia) / 86400_000)),
   146→        };
   147→      })
   148→      .sort((a, b) => b.dias - a.dias);
   149→
   150→    // 3. Tasa de averías por modelo (solo modelos con al menos 1 averiado)
   151→    const porModelo = new Map<string, { modelo: string; total: number; averiados: number }>();
   152→    for (const e of equipos) {
   153→      const k = e.modelo || "Sin modelo";
   154→      const cur = porModelo.get(k) ?? { modelo: k, total: 0, averiados: 0 };
   155→      cur.total += 1;
   156→      if (e.estado === "averiado") cur.averiados += 1;
   157→      porModelo.set(k, cur);
   158→    }
   159→    const tasaAverias = Array.from(porModelo.values())
   160→      .filter((m) => m.averiados > 0)
   161→      .map((m) => ({ ...m, tasa: m.total > 0 ? (m.averiados / m.total) * 100 : 0 }))
   162→      .sort((a, b) => b.tasa - a.tasa)
   163→      .slice(0, 5);
   164→
   165→    // 4. Top modelos más registrados
   166→    const topModelos = Array.from(porModelo.values())
   167→      .sort((a, b) => b.total - a.total)
   168→      .slice(0, 5);
   169→    const maxTopModelos = topModelos.length > 0 ? topModelos[0].total : 1;
   170→
   171→    return { averiadosConDias, sinUsoConDias, tasaAverias, topModelos, maxTopModelos };
   172→  }, [kpis, despachos, equipos]);
   173→
   174→  // Helper: tiempo relativo "hace X días/meses"
   175→  const fmtRelativo = (ts?: number) => {
   176→    if (!ts) return "—";
   177→    const diff = Date.now() - ts;
   178→    const dias = Math.floor(diff / 86400_000);
   179→    if (dias < 1) return "hoy";
   180→    if (dias === 1) return "ayer";
   181→    if (dias < 7) return `hace ${dias} días`;
   182→    if (dias < 30) return `hace ${Math.floor(dias / 7)} sem`;
   183→    const meses = Math.floor(dias / 30);
   184→    return meses === 1 ? "hace 1 mes" : `hace ${meses} meses`;
   185→  };
   186→
   187→  const goSeries = () => router.push("/series");
   188→
   189→  // ─── Count-up animations (Apple/iOS) para KPIs ───
   190→  const animCatalogo = useCountUp(equipos.length);
   191→  const animDisponibles = useCountUp(kpis.disponibles);
   192→  const animAveriados = useCountUp(kpis.averiados);
   193→  const animRetiro = useCountUp(kpis.enRetiro);
   194→  const animSinUso = useCountUp(kpis.sinUso);
   195→  const animModelos = useCountUp(kpis.modelosDistintos);
   196→
   197→  return (
   198→    <div className="select-text cursor-text px-6 py-4 anim-fade-in">
   199→      {/* Header */}
   200→      <header className="anim-slide-up mb-4 flex flex-wrap items-end justify-between gap-4">
   201→        <div>
   202→          <p className="text-[11px] font-medium uppercase tracking-[0.14em] text-muted-foreground">
   203→            {equipos.length} equipos · {models.length} modelos
   204→          </p>
   205→          <h1 className="mt-1 text-[28px] font-semibold tracking-tight text-foreground sm:text-[32px]">Equipos</h1>
   206→        </div>
   207→        <AuroraSearchInput
   208→          value={query}
   209→          onChange={setQuery}
   210→          placeholder="Buscar serie…"
   211→          placeholderFocused="Escribe para filtrar series…"
   212→          className="w-48"
   213→          iconSize="h-3.5 w-3.5"
   214→        />
   215→      </header>
   216→
   217→      {/* Pestañas: Equipos | Recomendaciones | Movimientos */}
   218→      <div className="anim-slide-up mb-3 flex items-center gap-1 border-b border-border">
   219→        {([
   220→          ["equipos", "Equipos", equipos.length],
   221→          ["recomendaciones", "Recomendaciones", kpis.averiados + kpis.sinUso + kpis.modelosDistintos],
   222→          ["movimientos", "Movimientos", despachos.length],
   223→        ] as const).map(([key, label, count]) => (
   224→          <button
   225→            key={key}
   226→            onClick={() => setTab(key)}
   227→            className={cn(
   228→              "relative inline-flex h-10 items-center gap-2 px-4 text-[13px] font-medium transition-colors",
   229→              tab === key ? "text-foreground" : "text-muted-foreground hover:text-foreground",
   230→            )}
   231→          >
   232→            {label}
   233→            <span
   234→              className={cn(
   235→                "rounded-full px-1.5 py-0.5 text-[10px] tabular-nums",
   236→                tab === key ? "bg-foreground text-background" : "bg-muted text-muted-foreground",
   237→              )}
   238→            >
   239→              {count}
   240→            </span>
   241→            {tab === key && (
   242→              <motion.span
   243→                layoutId="tab-indicator-equipos"
   244→                className="absolute inset-x-0 -bottom-px h-0.5 bg-foreground"
   245→                transition={{ type: "spring", stiffness: 400, damping: 30 }}
   246→              />
   247→            )}
   248→          </button>
   249→        ))}
   250→      </div>
   251→
   252→      <AnimatePresence mode="wait">
   253→      {/* ─── TAB: EQUIPOS (KPIs + chips + lista con paginación) ─── */}
   254→      {tab === "equipos" && (
   255→        <motion.div
   256→          key="equipos"
   257→          initial={{ opacity: 0, y: 8 }}
   258→          animate={{ opacity: 1, y: 0 }}
   259→          exit={{ opacity: 0, y: -8 }}
   260→          transition={{ duration: 0.25, ease: [0.22, 1, 0.36, 1] }}
   261→        >
   262→          {/* KPIs superiores — entrada escalonada con stagger delay */}
   263→          <div className="sticky-kpis mb-3 grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
   264→            {/* 1. Catálogo */}
   265→            <motion.div
   266→              whileHover={{ scale: 1.02 }}
   267→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   268→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   269→              style={{ animationDelay: "0ms" }}
   270→            >
   271→              <div className="flex items-center gap-1.5">
   272→                <Package className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   273→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Catálogo</p>
   274→              </div>
   275→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animCatalogo)}</p>
   276→            </motion.div>
   277→            {/* 2. Disponibles */}
   278→            <motion.div
   279→              whileHover={{ scale: 1.02 }}
   280→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   281→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   282→              style={{ animationDelay: "60ms" }}
   283→            >
   284→              <div className="flex items-center gap-1.5">
   285→                <Check className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   286→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Disponibles</p>
   287→              </div>
   288→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animDisponibles)}</p>
   289→            </motion.div>
   290→            {/* 3. Averiados — pulse dot si hay alerta */}
   291→            <motion.div
   292→              whileHover={{ scale: 1.02 }}
   293→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   294→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   295→              style={{ animationDelay: "120ms" }}
   296→            >
   297→              <div className="flex items-center gap-1.5">
   298→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Averiados</p>
   299→                {kpis.averiados > 0 && <span className="anim-pulse-dot h-2 w-2 rounded-full bg-destructive" />}
   300→              </div>
   301→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animAveriados)}</p>
   302→            </motion.div>
   303→            {/* 4. En retiro */}
   304→            <motion.div
   305→              whileHover={{ scale: 1.02 }}
   306→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   307→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   308→              style={{ animationDelay: "180ms" }}
   309→            >
   310→              <div className="flex items-center gap-1.5">
   311→                <Undo className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   312→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">En retiro</p>
   313→              </div>
   314→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animRetiro)}</p>
   315→            </motion.div>
   316→            {/* 5. Sin uso 60+d — pulse dot si hay alerta */}
   317→            <motion.div
   318→              whileHover={{ scale: 1.02 }}
   319→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   320→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   321→              style={{ animationDelay: "240ms" }}
   322→            >
   323→              <div className="flex items-center gap-1.5">
   324→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Sin uso 60+d</p>
   325→                {kpis.sinUso > 0 && <span className="anim-pulse-dot h-2 w-2 rounded-full bg-amber-500" />}
   326→              </div>
   327→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animSinUso)}</p>
   328→            </motion.div>
   329→            {/* 6. Modelos */}
   330→            <motion.div
   331→              whileHover={{ scale: 1.02 }}
   332→              transition={{ type: "spring", stiffness: 300, damping: 20 }}
   333→              className="press-card anim-slide-up rounded-xl border border-border bg-card p-3 text-card-foreground"
   334→              style={{ animationDelay: "300ms" }}
   335→            >
   336→              <div className="flex items-center gap-1.5">
   337→                <Boxes className="h-3 w-3 text-muted-foreground" {...ICON_PROPS} />
   338→                <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground">Modelos</p>
   339→              </div>
   340→              <p className="tabular mt-1 text-2xl font-bold tabular-nums text-foreground">{fmtNum(animModelos)}</p>
   341→            </motion.div>
   342→          </div>
   343→
   344→          {/* Chips de filtro por estado */}
   345→          <div className="anim-slide-up mb-3 flex flex-wrap items-center gap-1">
   346→            <Filter className="mr-1 h-3.5 w-3.5 text-muted-foreground" {...ICON_PROPS} />
   347→            <FilterChip active={estadoFilter === "todos"} onClick={() => setEstadoFilter("todos")} label="Todos" count={equipos.length} />
   348→            {ESTADOS.map((est) => {
   349→              const n = equipos.filter((e) => e.estado === est).length;
   350→              return <FilterChip key={est} active={estadoFilter === est} onClick={() => setEstadoFilter(est)} label={ESTADO_META[est].short} count={n} />;
   351→            })}
   352→          </div>
   353→
   354→          {/* Lista de modelos con paginación */}
   355→          {models.length === 0 ? (
   356→            <p className="py-12 text-center text-[13px] text-muted-foreground">
   357→              {equipos.length === 0 ? "No hay equipos registrados. Los equipos se añaden automáticamente al recibir una guía de remisión SUNAT en la página de Recepciones." : "Sin coincidencias."}
   358→            </p>
   359→          ) : (
   360→            <>
   361→              <div className="divide-y divide-border border-y border-border">
   362→                {paginaModels.map(([modelo, items], idx) => {
   363→                  // ─── Antigüedad: contar equipos disponibles por nivel ───
   364→                  const disp = items.filter((e) => e.estado === "disponible");
   365→                  const ant = disp.map((e) => calcularAntiguedad(e.createdAt));
   366→                  const antAdv = ant.filter((a) => a.nivel === "advertencia").length;
   367→                  const antCrit = ant.filter((a) => a.nivel === "critico").length;
   368→                  const antOld = ant.filter((a) => a.nivel === "antiguo").length;
   369→                  const antTotal = antAdv + antCrit + antOld;
   370→                  return (
   371→                    <div
   372→                      key={modelo}
   373→                      className="anim-fade-in flex items-center gap-4 py-4"
   374→                      style={{ animationDelay: `${idx * 25}ms` }}
   375→                    >
   376→                      <div className="min-w-0 flex-1">
   377→                        <p className="text-[15px] font-medium text-foreground">{modelo}</p>
   378→                        <p className="mt-0.5 text-[12px] text-muted-foreground">
   379→                          {items.length} equipo(s) · {disp.length} disponibles
   380→                          {items.some((e) => e.ubicacion) && (
   381→                            <span className="ml-1.5 inline-flex items-center gap-1 text-[11px]">
   382→                              <MapPin className="h-3 w-3" {...ICON_PROPS} />
   383→                              {[...new Set(items.map((e) => e.ubicacion).filter(Boolean))].slice(0, 3).join(", ")}
   384→                              {[...new Set(items.map((e) => e.ubicacion).filter(Boolean))].length > 3 && "…"}
   385→                            </span>
   386→                          )}
   387→                          {antTotal > 0 && (
   388→                            <span className="ml-1.5 inline-flex items-center gap-1 text-[11px]">
   389→                              <Clock className="h-3 w-3" {...ICON_PROPS} />
   390→                              {antTotal} con antigüedad
   391→                            </span>
   392→                          )}
   393→                        </p>
   394→                      </div>
   395→                      <div className="hidden items-center gap-3 sm:flex">
   396→                        {ESTADOS.map((est) => {
   397→                          const n = items.filter((e) => e.estado === est).length;
   398→                          if (n === 0) return null;
   399→                          return (
   400→                            <span key={est} className="flex items-center gap-1.5 text-[12px] text-muted-foreground">
   401→                              <EstadoIcon name={ESTADO_META[est].icon} className="h-3 w-3" />
   402→                              {n}
   403→                            </span>
   404→                          );
   405→                        })}
   406→                        {/* Indicadores de antigüedad (solo si hay equipos disponibles) */}
   407→                        {antAdv > 0 && (
   408→                          <span className={cn("inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-[10px] font-medium", ANTIGUEDAD_META.advertencia.bg, ANTIGUEDAD_META.advertencia.text)}>
   409→                            <span className={cn("h-1.5 w-1.5 rounded-full", ANTIGUEDAD_META.advertencia.dot)} />
   410→                            {antAdv} advert.
   411→                          </span>
   412→                        )}
   413→                        {antCrit > 0 && (
   414→                          <span className={cn("inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-[10px] font-medium", ANTIGUEDAD_META.critico.bg, ANTIGUEDAD_META.critico.text)}>
   415→                            <span className={cn("h-1.5 w-1.5 rounded-full", ANTIGUEDAD_META.critico.dot)} />
   416→                            {antCrit} crít.
   417→                          </span>
   418→                        )}
   419→                        {antOld > 0 && (
   420→                          <span className={cn("inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-[10px] font-medium", ANTIGUEDAD_META.antiguo.bg, ANTIGUEDAD_META.antiguo.text)}>
   421→                            <span className={cn("h-1.5 w-1.5 rounded-full", ANTIGUEDAD_META.antiguo.dot)} />
   422→                            {antOld} antiguos
   423→                          </span>
   424→                        )}
   425→                      </div>
   426→                      <span className="text-[15px] font-semibold tabular-nums text-foreground">{items.length}</span>
   427→                      <motion.button
   428→                        onClick={goSeries}
   429→                        whileTap={{ scale: 0.97 }}
   430→                        transition={{ type: "spring", stiffness: 300, damping: 20 }}
