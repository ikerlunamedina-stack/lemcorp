   146→   146→function detectarModeloPorPrefijo(serie: string): string | null {
   147→   147→  const s = serie.trim().toUpperCase();
   148→   148→  if (!s) return null;
   149→   149→  for (const regla of REGLAS_PREFIJO) {
   150→   150→    if (s.startsWith(regla.prefijo.toUpperCase())) return regla.modelo;
   151→   151→  }
   152→   152→  return null;
   153→   153→}

   155→   155→export const useStore = create<StoreState>()(
   156→   156→  persist(
   157→   157→    (set, get) => ({
   158→   158→      products: [],
   159→   159→      equipos: [],
   160→   160→      entradas: [],
   161→   161→      despachos: [],
   162→   162→      notas: [],
   163→   163→      recordatorios: [],
   164→   164→      notificaciones: [],
   165→   165→      miembros: [],
   166→   166→      empresa: { ...DEFAULT_EMPRESA },
   167→   167→      settings: { ...DEFAULT_SETTINGS },
   168→   168→
   169→   169→      activeView: "dashboard",
   170→   170→
   171→   171→      // pistoleo
   172→   172→      pistoleoCampo: "serie",
   173→   173→      pistoleoModelo: "",
   174→   174→      pistoleoEstado: "disponible",
   175→   175→      pistoleoFilas: [],
   176→   176→
   177→   177→      // ─── Navegación ───
   178→   178→      setActiveView: (v) => set({ activeView: v }),
   179→   179→
   180→   180→      // ─── Pistoleo ───
   181→   181→      setPistoleoConfig: (patch) => set({ ...patch }),
   182→   182→      addPistoleoFila: (valores) =>
   183→   183→        set({
   184→   184→          pistoleoFilas: [
   185→   185→            { id: uid(), valores: valores.map((v) => v.trim()), timestamp: Date.now() },
   186→   186→            ...get().pistoleoFilas,
   187→   187→          ],
   188→   188→        }),
   189→   189→      deletePistoleoFila: (id) =>
   190→   190→        set({ pistoleoFilas: get().pistoleoFilas.filter((f) => f.id !== id) }),
   191→   191→      clearPistoleoFilas: () => set({ pistoleoFilas: [] }),
   192→   192→
   193→   193→      confirmarPistoleo: () => {
   194→   194→        const filas = get().pistoleoFilas;
   195→   195→        if (filas.length === 0) return { ok: false, msg: "No hay series para guardar.", count: 0 };
   196→   196→        const { pistoleoModelo, pistoleoEstado } = get();
   197→   197→        let count = 0;
   198→   198→        const nuevos: Equipment[] = [];
   199→   199→        const existentes = new Set(get().equipos.map((e) => e.serie.trim().toLowerCase()));
   200→   200→        const fechasNow = Date.now();
   201→   201→        for (const f of filas) {
   202→   202→          const serie = (f.valores[0] ?? "").trim();
   203→   203→          if (!serie) continue;
   204→   204→          if (existentes.has(serie.toLowerCase())) continue;
   205→   205→          const modelo = pistoleoModelo.trim() || detectarModeloPorPrefijo(serie) || "SIN MODELO";
   206→   206→          nuevos.push({
   207→   207→            id: uid(),
   208→   208→            serie,
   209→   209→            modelo,
   210→   210→            estado: pistoleoEstado,
   211→   211→            ubicacion: "Almacén HUB",
   212→   212→            createdAt: fechasNow,
   213→   213→            updatedAt: fechasNow,
   214→   214→          });
   215→   215→          existentes.add(serie.toLowerCase());
   216→   216→          count++;
   217→   217→        }
   218→   218→        if (count === 0) {
   219→   219→          return { ok: false, msg: "Todas las series ya estaban registradas.", count: 0 };
   220→   220→        }
   221→   221→        set({ equipos: [...nuevos, ...get().equipos], pistoleoFilas: [] });
   222→   222→        return {
   223→   223→          ok: true,
   224→   224→          msg: `${count} equipo(s) guardado(s) correctamente.`,
   225→   225→          count,
   226→   226→        };
   227→   227→      },
   228→   228→
   229→   229→      // ─── Inventario ───
   230→   230→      findProductBySku: (sku) => {
   231→   231→        const norm = sku.trim().toLowerCase();
   232→   232→        return get().products.find((p) => p.sku.trim().toLowerCase() === norm) ?? null;
   233→   233→      },
   234→   234→
   235→   235→      addProduct: (sku, name, quantity, minStock, udm) => {
   236→   236→        const skuTrim = sku.trim();
   237→   237→        if (!skuTrim || !name.trim()) return null;
   238→   238→        if (get().findProductBySku(skuTrim)) return null;
   239→   239→        const p: Product = {
   240→   240→          id: uid(),
   241→   241→          sku: skuTrim,
   242→   242→          name: name.trim(),
   243→   243→          quantity: quantity || 0,
   244→   244→          minStock,
   245→   245→          udm,
   246→   246→          createdAt: Date.now(),
   247→   247→          updatedAt: Date.now(),
   248→   248→        };
   249→   249→        set({ products: [...get().products, p] });
   250→   250→        return p.id;
   251→   251→      },
   252→   252→
   253→   253→      updateProduct: (id, data) =>
   254→   254→        set({
   255→   255→          products: get().products.map((p) =>
   256→   256→            p.id === id ? { ...p, ...data, updatedAt: Date.now() } : p
   257→   257→          ),
   258→   258→        }),
   259→   259→
   260→   260→      deleteProduct: (id) =>
   261→   261→        set({ products: get().products.filter((p) => p.id !== id) }),
   262→   262→
   263→   263→      importProductsBulk: (items) => {
   264→   264→        let ok = 0;
   265→   265→        let dup = 0;
   266→   266→        const nuevos: Product[] = [];
   267→   267→        const existSkus = new Set(get().products.map((p) => p.sku.trim().toLowerCase()));
   268→   268→        for (const it of items) {
   269→   269→          const sku = it.sku.trim();
   270→   270→          if (!sku || !it.name.trim()) continue;
   271→   271→          if (existSkus.has(sku.toLowerCase())) { dup++; continue; }
   272→   272→          existSkus.add(sku.toLowerCase());
   273→   273→          nuevos.push({
   274→   274→            id: uid(),
   275→   275→            sku,
   276→   276→            name: it.name.trim(),
   277→   277→            quantity: it.quantity || 0,
   278→   278→            minStock: it.minStock,
   279→   279→            udm: it.udm,
   280→   280→            createdAt: Date.now(),
   281→   281→            updatedAt: Date.now(),
   282→   282→          });
   283→   283→          ok++;
   284→   284→        }
   285→   285→        if (nuevos.length > 0) set({ products: [...get().products, ...nuevos] });
   286→   286→        return { ok, dup };
   287→   287→      },
   288→   288→
   289→   289→      // ─── Entradas (formato SKU*cantidad por línea) ───
   290→   290→      registrarEntrada: (input) => {
   291→   291→        const lines = input.split("\n").filter((l) => l.trim());
   292→   292→        let count = 0;
   293→   293→        const newEntradas: Entrada[] = [];
   294→   294→        for (const line of lines) {
   295→   295→          const parts = line.trim().split("*");
   296→   296→          if (parts.length < 2) continue;
   297→   297→          const sku = parts[0].trim();
   298→   298→          const cantidad = parseInt(parts[1].trim(), 10);
   299→   299→          if (!sku || isNaN(cantidad) || cantidad <= 0) continue;
   300→   300→          const product = get().findProductBySku(sku);
   301→   301→          if (product) {
   302→   302→            set({
   303→   303→              products: get().products.map((p) =>
   304→   304→                p.id === product.id
   305→   305→                  ? { ...p, quantity: p.quantity + cantidad, updatedAt: Date.now() }
   306→   306→                  : p
   307→   307→              ),
   308→   308→            });
   309→   309→          }
   310→   310→          newEntradas.push({
   311→   311→            id: uid(),
   312→   312→            fecha: Date.now(),
   313→   313→            sku,
   314→   314→            producto: product?.name ?? sku,
   315→   315→            cantidad,
   316→   316→          });
   317→   317→          count++;
   318→   318→        }
   319→   319→        if (count > 0) set({ entradas: [...newEntradas, ...get().entradas] });
   320→   320→        return {
   321→   321→          ok: count > 0,
   322→   322→          msg: count > 0
   323→   323→            ? `${count} entrada(s) registrada(s)`
   324→   324→            : "Formato incorrecto. Usa: SKU*cantidad (ej: 1066990*100)",
   325→   325→          count,
   326→   326→        };
   327→   327→      },
   328→   328→
   329→   329→      deleteEntrada: (id) => {
   330→   330→        const ent = get().entradas.find((e) => e.id === id);
   331→   331→        if (!ent) return;
   332→   332→        const product = get().findProductBySku(ent.sku);
   333→   333→        if (product) {
   334→   334→          set({
   335→   335→            products: get().products.map((p) =>
   336→   336→              p.id === product.id
   337→   337→                ? { ...p, quantity: Math.max(0, p.quantity - ent.cantidad), updatedAt: Date.now() }
   338→   338→                : p
   339→   339→            ),
   340→   340→          });
   341→   341→        }
   342→   342→        set({ entradas: get().entradas.filter((e) => e.id !== id) });
   343→   343→      },
   344→   344→
   345→   345→      // ─── Despachos (valida SKU y stock, descuenta inventario) ───
   346→   346→      registrarDespacho: ({ sku, cantidad, tecnico, destino, observacion, fecha }) => {
   347→   347→        const skuTrim = sku.trim();
   348→   348→        const product = get().findProductBySku(skuTrim);
   349→   349→        if (!product) {
   350→   350→          return { ok: false, msg: `SKU "${skuTrim}" no encontrado en el catálogo.` };
   351→   351→        }
   352→   352→        if (!Number.isFinite(cantidad) || cantidad <= 0) {
   353→   353→          return { ok: false, msg: "La cantidad debe ser mayor a 0." };
   354→   354→        }
   355→   355→        if (product.quantity < cantidad) {
   356→   356→          return {
   357→   357→            ok: false,
   358→   358→            msg: `Stock insuficiente. Disponible: ${product.quantity}, solicitado: ${cantidad}.`,
   359→   359→          };
   360→   360→        }
   361→   361→        const d: Despacho = {
   362→   362→          id: uid(),
   363→   363→          fecha: fecha || Date.now(),
   364→   364→          sku: product.sku,
   365→   365→          producto: product.name,
   366→   366→          cantidad,
   367→   367→          tecnico: tecnico?.trim() || undefined,
   368→   368→          destino: destino?.trim() || undefined,
   369→   369→          observacion: observacion?.trim() || undefined,
   370→   370→        };
   371→   371→        set({
   372→   372→          despachos: [d, ...get().despachos],
   373→   373→          products: get().products.map((p) =>
   374→   374→            p.id === product.id
   375→   375→              ? { ...p, quantity: p.quantity - cantidad, updatedAt: Date.now() }
   376→   376→              : p
   377→   377→          ),
   378→   378→        });
   379→   379→        return {
   380→   380→          ok: true,
   381→   381→          msg: `Despacho registrado: ${cantidad} × ${product.name} → stock: ${product.quantity - cantidad}`,
   382→   382→        };
   383→   383→      },
   384→   384→
   385→   385→      // Registro masivo de despachos (para Excel importado)
   386→   386→      registrarDespachosBulk: (inputDespachos) => {
   387→   387→        let ok = 0;
   388→   388→        let fail = 0;
   389→   389→        let totalUnidades = 0;
   390→   390→        const fails: string[] = [];
   391→   391→        const nuevosDespachos: Despacho[] = [];
   392→   392→        const productosActualizados = new Map<string, number>();
   393→   393→
   394→   394→        for (const item of inputDespachos) {
   395→   395→          const skuTrim = item.sku.trim();
   396→   396→          const product = get().findProductBySku(skuTrim);
   397→   397→          if (!product) {
   398→   398→            fail++;
   399→   399→            fails.push(`SKU "${skuTrim}" no encontrado`);
   400→   400→            continue;
   401→   401→          }
   402→   402→          if (!Number.isFinite(item.cantidad) || item.cantidad <= 0) {
   403→   403→            fail++;
   404→   404→            fails.push(`Cantidad inválida para ${skuTrim}`);
   405→   405→            continue;
   406→   406→          }
   407→   407→          // Calcular stock disponible considerando despachos anteriores del mismo bulk
   408→   408→          const stockActual = productosActualizados.get(product.id) ?? product.quantity;
   409→   409→          if (stockActual < item.cantidad) {
   410→   410→            fail++;
   411→   411→            fails.push(`Stock insuficiente para ${product.name} (disp: ${stockActual}, solicitado: ${item.cantidad})`);
   412→   412→            continue;
   413→   413→          }
   414→   414→          productosActualizados.set(product.id, stockActual - item.cantidad);
   415→   415→          nuevosDespachos.push({
   416→   416→            id: uid(),
   417→   417→            fecha: item.fecha || Date.now(),
   418→   418→            sku: product.sku,
   419→   419→            producto: product.name,
   420→   420→            cantidad: item.cantidad,
   421→   421→            tecnico: item.tecnico?.trim() || undefined,
   422→   422→            destino: item.destino?.trim() || undefined,
   423→   423→            observacion: item.observacion?.trim() || undefined,
   424→   424→          });
   425→   425→          ok++;
   426→   426→          totalUnidades += item.cantidad;
   427→   427→        }
   428→   428→
   429→   429→        if (nuevosDespachos.length > 0) {
   430→   430→          // Ordenar por fecha (más reciente primero)
   431→   431→          nuevosDespachos.sort((a, b) => b.fecha - a.fecha);
   432→   432→          set({
   433→   433→            despachos: [...nuevosDespachos, ...get().despachos],
   434→   434→            products: get().products.map((p) => {
   435→   435→              const newQty = productosActualizados.get(p.id);
   436→   436→              return newQty !== undefined ? { ...p, quantity: newQty, updatedAt: Date.now() } : p;
   437→   437→            }),
   438→   438→          });
   439→   439→        }
   440→   440→        return { ok, fail, fails, totalUnidades };
   441→   441→      },
   442→   442→
   443→   443→      deleteDespacho: (id) => {
   444→   444→        const d = get().despachos.find((x) => x.id === id);
   445→   445→        if (!d) return;
   446→   446→        const product = get().findProductBySku(d.sku);
   447→   447→        if (product) {
   448→   448→          set({
   449→   449→            products: get().products.map((p) =>
   450→   450→              p.id === product.id
